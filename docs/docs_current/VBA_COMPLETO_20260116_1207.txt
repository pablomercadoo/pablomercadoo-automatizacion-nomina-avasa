Private Sub Workbook_Open()

    ' 2) Leer config IsTemplate

Private Sub FillLocationFieldsIfMissing()
    On Error GoTo EH

    Dim locCode As String, locName As String, locDisp As String
    locCode = Trim$(GetConfig("LocationCode", ""))
    locName = Trim$(GetConfig("LocationName", ""))
    locDisp = Trim$(GetConfig("LocationDisplay", ""))

    If Len(locCode) = 0 Then Exit Sub

    'Si ya está lleno, no hacemos nada
    If Len(locName) > 0 And Len(locDisp) > 0 Then Exit Sub

    'Intentar sacar de tblLocaciones
    Dim ws As Worksheet, lo As ListObject, lr As ListRow
    Set ws = ThisWorkbook.Worksheets("Locaciones")
    Set lo = ws.ListObjects("tblLocaciones")

    Dim idxCode As Long, idxName As Long, idxDisp As Long
    idxCode = lo.ListColumns("LocationCode").Index

    On Error Resume Next
    idxName = lo.ListColumns("LocationName").Index
    idxDisp = lo.ListColumns("LocationDisplay").Index
    On Error GoTo EH

    For Each lr In lo.ListRows
        If UCase$(Trim$(CStr(lr.Range.Cells(1, idxCode).value))) = UCase$(locCode) Then

            If Len(locName) = 0 Then
                If idxName > 0 Then locName = Trim$(CStr(lr.Range.Cells(1, idxName).value))
            End If

            If Len(locDisp) = 0 Then
                If idxDisp > 0 Then locDisp = Trim$(CStr(lr.Range.Cells(1, idxDisp).value))
            End If

            Exit For
        End If
    Next lr

    'Fallbacks
    If Len(locName) = 0 Then locName = locCode
    If Len(locDisp) = 0 Then locDisp = locName

    'Guardar
    SetConfig "LocationCode", locCode
    SetConfig "LocationName", locName
    SetConfig "LocationDisplay", locDisp
    Exit Sub

EH:
    'Si algo falla, al menos fallback directo
    If Len(locCode) > 0 Then
        SetConfig "LocationName", locCode
        SetConfig "LocationDisplay", locCode
    End If
End Sub

Private Function IsWebPath(ByVal p As String) As Boolean
    Dim s As String
    s = LCase$(Trim$(p))
    IsWebPath = (Left$(s, 4) = "http") Or (InStr(s, "sharepoint") > 0) Or (InStr(s, "onedrive") > 0)
End Function

'===================================================
'  ÚLTIMO DÍA DEL MES
'===================================================
Private Function UltimoDiaMes(anio As Long, mes As Long) As Long
    UltimoDiaMes = Day(DateSerial(anio, mes + 1, 0))
End Function


'===================================================
'  OBTENER RANGO DE DÍAS DEL PERIODO
'===================================================
Private Sub ObtenerRangoPeriodo( _
        ByVal anio As Long, _
        ByVal mes As Long, _
        ByVal tipoPeriodo As String, _
        ByVal numPeriodo As Long, _
        ByRef diaIni As Long, _
        ByRef diaFin As Long)

    Dim uDia As Long
    uDia = UltimoDiaMes(anio, mes)

    Select Case UCase(tipoPeriodo)

        Case "SEMANAL"
            Select Case numPeriodo
                Case 1: diaIni = 1:  diaFin = 7
                Case 2: diaIni = 8:  diaFin = 14
                Case 3: diaIni = 15: diaFin = 21
                Case 4: diaIni = 22: diaFin = uDia
            End Select

        Case "QUINCENAL"
            Select Case numPeriodo
                Case 1: diaIni = 1:  diaFin = 15
                Case 2: diaIni = 16: diaFin = uDia
            End Select
    End Select
End Sub


'===================================================
'  CONFIGURAR LABELS/COMBOS DE DÍA Y TEXTO DEL PERIODO
'===================================================
Private Sub ConfigurarPeriodo()
    Dim numDias As Long
    Dim i As Long
    Dim dia As Long
    Dim fIni As Date, fFin As Date
    Dim nombreMes As String

    If gAnio = 0 Or gMes = 0 Or gTipoPeriodo = "" Or gPeriodo = 0 Then Exit Sub

    ObtenerRangoPeriodo gAnio, gMes, gTipoPeriodo, gPeriodo, gDiaIni, gDiaFin
    numDias = gDiaFin - gDiaIni + 1

    fIni = DateSerial(gAnio, gMes, gDiaIni)
    fFin = DateSerial(gAnio, gMes, gDiaFin)
    nombreMes = UCase(Format(fIni, "mmmm"))

    lblPeriodo.Caption = Format(fIni, "dd") & "-" & _
                         Format(fFin, "dd") & " " & _
                         nombreMes & " " & gAnio

    For i = 1 To 16
        With Me.Controls("lblDia" & i)
            If i <= numDias Then
                dia = gDiaIni + i - 1
                .Caption = CStr(dia)
                .Visible = True
            Else
                .Caption = ""
                .Visible = False
            End If
        End With

        With Me.Controls("cboDia" & i)
            If i <= numDias Then
                .Visible = True
            Else
                .value = ""
                .Visible = False
            End If
        End With
    Next i
End Sub


Private Sub txtNumEmpleado_AfterUpdate()

If mSilenciarAfterUpdate Or EnEdicion Then Exit Sub

    On Error GoTo EH

    Dim numEmp As String
    Dim resp As VbMsgBoxResult
    Dim encontrado As Boolean

    numEmp = Trim$(CStr(txtNumEmpleado.value))

    If Len(numEmp) = 0 Then Exit Sub
    If Not IsNumeric(numEmp) Then
        MsgBox "El número de empleado debe ser numérico.", vbExclamation
        txtNumEmpleado.value = ""
        Exit Sub
    End If

    encontrado = False

    ' 1) Primero intenta cargar de Empleados (OFICIAL)
    If CargarEmpleado_Oficial(numEmp) Then
        mAltaTemporal = False
        AplicarModoEmpleadoTemporal False
        encontrado = True
    Else
        ' 2) Si no existe oficial, intenta cargar de Empleados_Temp (TEMPORAL)
        If CargarEmpleado_Temporal(numEmp) Then
            mAltaTemporal = True
            AplicarModoEmpleadoTemporal True
            encontrado = True
        End If
    End If

    ' Si se encontró (oficial o temporal), ofrecer cargar incidencias existentes
    If encontrado Then
        If ExisteIncidenciasEmpleadoPeriodo(CLng(numEmp)) Then
            resp = MsgBox( _
                "Ya existen incidencias para este empleado en este periodo." & vbCrLf & _
                "¿Deseas cargarlas para editar?", _
                vbQuestion + vbYesNo + vbDefaultButton1, "Editar incidencias")

            If resp = vbYes Then
                Call CargarDesdeBD(CLng(numEmp))
            End If
        End If

        Exit Sub
    End If

    ' 3) Si no existe en ningún lado, ofrece alta temporal
    If MsgBox("El empleado " & numEmp & " no existe en la base oficial." & vbCrLf & vbCrLf & _
              "¿Deseas darlo de ALTA TEMPORAL para este periodo?", _
              vbQuestion + vbYesNo, "Alta temporal") = vbYes Then

        mAltaTemporal = True
        LimpiarCamposEmpleado
        txtNumEmpleado.value = numEmp
        AplicarModoEmpleadoTemporal True
        txtNombre.SetFocus

    Else
        txtNumEmpleado.value = ""
        LimpiarCamposEmpleado
        mAltaTemporal = False
        AplicarModoEmpleadoTemporal False
    End If

    Exit Sub

EH:
    MsgBox "Error en txtNumEmpleado_AfterUpdate: " & err.Number & " - " & err.Description, vbExclamation

End Sub

'===================================================
'  Validar que un código de incidencia esté en glosario
'===================================================
Private Function EsCodigoValido(ByVal cod As String) As Boolean
    Dim k As Long
    cod = UCase$(Trim$(cod))

    'Vacío siempre es válido (sin incidencia)
    If cod = "" Then
        EsCodigoValido = True
        Exit Function
    End If

    For k = LBound(tiposInc) To UBound(tiposInc)
        If cod = tiposInc(k) Then
            EsCodigoValido = True
            Exit Function
        End If
    Next k

    EsCodigoValido = False
End Function
'Devuelve el CODIGO real del combo (col 1)
Private Function GetCodigoFromCombo(ByVal cbo As MSForms.ComboBox) As String
    On Error GoTo safe

    If cbo.ListIndex < 0 Then
        GetCodigoFromCombo = ""
    Else
        'Columna 1 (index 0) = codigo
        GetCodigoFromCombo = UCase$(Trim$(CStr(cbo.List(cbo.ListIndex, 0))))
    End If
    Exit Function

safe:
    GetCodigoFromCombo = UCase$(Trim$(CStr(cbo.value)))
End Function

    ' Guardar overrides de Puesto / Actividad
    ' (nivel EMPLEADO + PERIODO)
    '=====================================
    paso = "UpsertOverridesEmpleadoPeriodo"
    modReporteIncidencias.UpsertOverridesEmpleadoPeriodo _
        CLng(txtNumEmpleado.value), _
        Trim$(cmbPuesto.value), _
        Trim$(cmbActividad.value)

    '=====================================
    ' Refrescar matriz (sin parpadeo)
    '=====================================
    paso = "GenerarMatrizPeriodoActual"
    modReporteIncidencias.GenerarMatrizPeriodoActual

    DoEvents

    MsgBox "Incidencias guardadas/actualizadas.", vbInformation

    '=====================================
    ' Cerrar SOLO si es edición
    '=====================================
    If EnEdicion Then
        Unload Me
        GoTo FIN
    End If

    '-------------------------------------
    ' Si NO es edición, limpiar para nueva captura
    '-------------------------------------
    paso = "LimpiarForm"
    LimpiarIncidencias
    txtAdicional.value = ""
    txtObservaciones.value = ""
    On Error Resume Next
    txtBonoComedor.value = ""
    On Error GoTo EH

    txtNumEmpleado.value = ""
    txtUsuarioCars.value = ""
    txtDriverCars.value = ""
    cmbPuesto.value = ""
    cmbActividad.value = ""
    txtNombre.value = ""

    EnEdicion = False

FIN:
    '========================
    ' TURBO OFF (SIEMPRE)
    '========================
    Application.Calculation = calc0
    Application.EnableEvents = ev0
    Application.ScreenUpdating = su0
    Exit Sub

EH:
    'Intentar reactivar Excel sí o sí
    On Error Resume Next
    Application.Calculation = calc0
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    On Error GoTo 0

    MsgBox "Error al guardar (" & paso & "): " & err.Number & " - " & err.Description, vbCritical
End Sub

Private Sub MoverCodigoBAlFinal(ByRef arr() As String)
    Dim i As Long, j As Long
    Dim tmp As String
    Dim idxB As Long

    On Error GoTo FIN
    idxB = -1

    If (LBound(arr) > UBound(arr)) Then Exit Sub

    For i = LBound(arr) To UBound(arr)
        If UCase$(Trim$(arr(i))) = "B" Then
            idxB = i
            Exit For
        End If
    Next i

    If idxB = -1 Then Exit Sub

    tmp = arr(idxB)
    For j = idxB To UBound(arr) - 1
        arr(j) = arr(j + 1)
    Next j
    arr(UBound(arr)) = tmp

FIN:
End Sub

'===================================================
'  AJUSTA UN CÓDIGO SEGÚN EL DÍA REAL
'  - Domingo => PD (salvo B o vacío)
'  - Festivo => DF (salvo B o vacío)
'  - Normal => si trae PD/DF lo convierte a X (si existe), si no a vacío
'===================================================
Private Function AjustarCodigoPorFecha(ByVal cod As String, ByVal f As Date) As String
    Dim tipoDia As String
    cod = UCase$(Trim$(cod))

    If cod = "" Then
        AjustarCodigoPorFecha = ""
        Exit Function
    End If

    ' BAJA siempre válida
    If cod = "B" Then
        AjustarCodigoPorFecha = "B"
        Exit Function
    End If

    tipoDia = modCalendario.GetDiaTipo(f) ' NORMAL / PD / DF

    '--- Si es domingo o feriado, NO permitir X
    If (tipoDia = "PD" Or tipoDia = "DF") And cod = "X" Then
        AjustarCodigoPorFecha = ""
        Exit Function
    End If

    '--- Si es día normal, NO permitir PD ni DF
    If tipoDia = "NORMAL" And (cod = "PD" Or cod = "DF") Then
        AjustarCodigoPorFecha = ""
        Exit Function
    End If

    AjustarCodigoPorFecha = cod
End Function

'===================================================
'  CARGA LA LISTA DEL COMBO SEGÚN EL DÍA REAL
'  UI: muestra CODIGO + DESCRIPCIÓN (2 columnas)
'  Valor real: guarda CÓDIGO (col 1)
'===================================================
Private Sub CargarListaComboSegunDia(ByVal cbo As MSForms.ComboBox, ByVal f As Date)

    Dim k As Long
    Dim cod As String
    Dim tipoDia As String
    Dim desc As String
    Dim rowIx As Long

    cbo.Clear

    '2 columnas: [CODIGO] [DESCRIPCIÓN]
    cbo.ColumnCount = 2
    cbo.BoundColumn = 1               'Value = CODIGO (col 1)
    cbo.ColumnWidths = "26 pt;200 pt" 'ajusta a gusto
    cbo.ListWidth = 260               'ancho del desplegable (ajusta)

    'Primera opción vacía
    cbo.AddItem ""
    rowIx = cbo.ListCount - 1
    cbo.List(rowIx, 1) = ""           'desc vacía

    tipoDia = modCalendario.GetDiaTipo(f) ' "NORMAL" / "PD" / "DF"

    If IsArray(tiposInc) Then
        For k = LBound(tiposInc) To UBound(tiposInc)

            cod = UCase$(Trim$(CStr(tiposInc(k))))
            If cod = "" Then GoTo NextK

            '--- NO mostrar PD/DF en días normales
            If tipoDia = "NORMAL" Then
                If cod = "PD" Or cod = "DF" Then GoTo NextK
            End If

            '--- NO mostrar X en domingos o feriados
            If (tipoDia = "PD" Or tipoDia = "DF") Then
                If cod = "X" Then GoTo NextK
            End If

            desc = modCatalogoIncidencias.DescripcionDeCodigo(cod)

            'Agregar fila: col1=COD, col2=DESC
            cbo.AddItem cod
            rowIx = cbo.ListCount - 1
            cbo.List(rowIx, 1) = desc

NextK:
        Next k
    End If
End Sub

'===================================================
'  APLICA REGLAS A TODOS LOS COMBOS VISIBLES
'  1) reconstruye lista según el día real
'  2) corrige selección si está “fuera de regla”
'===================================================
Private Sub AplicarReglasDias_EnCombos()
    Dim i As Long, diaReal As Long
    Dim f As Date
    Dim codAntes As String, codFinal As String
    Dim cbo As MSForms.ComboBox

    For i = 1 To 16
        If Me.Controls("cboDia" & i).Visible Then

            Set cbo = Me.Controls("cboDia" & i)

            diaReal = CLng(val(Me.Controls("lblDia" & i).Caption))
            If diaReal <= 0 Then GoTo NextI

            f = DateSerial(gAnio, gMes, diaReal)

            '1) Guardar CÓDIGO actual (NO Value)
            codAntes = GetCodigoFromCombo(cbo)
            codAntes = modCatalogoIncidencias.CanonizarCodigo(codAntes)

            '2) Reconstruir lista permitida (esto hace Clear)
            CargarListaComboSegunDia cbo, f

            '3) Ajustar y reponer por reglas del día
            codFinal = AjustarCodigoPorFecha(codAntes, f)

            If codFinal <> "" Then
                If modCatalogoIncidencias.EsCodigoValido(codFinal) Then
                    SetComboByCodigo cbo, codFinal
                Else
                    cbo.value = ""
                End If
            Else
                cbo.value = ""
            End If
        End If
NextI:
    Next i
End Sub

'====================================================
' Carga el empleado desde BDIncidencias_Local para el periodo actual
' (gLoc, gAnio, gMes, gTipoPeriodo, gPeriodo ya deben estar seteados)
'====================================================
Public Sub CargarDesdeBD(ByVal numEmp As Long)

    Dim wsBD As Worksheet
    Dim lastRow As Long, i As Long
    Dim diaBD As Long
    Dim cod As String
    Dim dictCod As Object
    Dim tipoBD As String, tipoG As String
    Dim encontradoEmp As Boolean

    On Error GoTo EH

    '----- modo edición -----
    Me.EnEdicion = True

    '----- silenciar AfterUpdate mientras llenamos -----
    mSilenciarAfterUpdate = True

    '----- limpiar UI antes de cargar -----
    txtNumEmpleado.value = CStr(numEmp)

    txtUsuarioCars.value = ""
    txtDriverCars.value = ""
    cmbPuesto.value = ""
    cmbActividad.value = ""
    txtNombre.value = ""
    txtAdicional.value = ""
    txtObservaciones.value = ""
    On Error Resume Next
    txtBonoComedor.value = ""
    On Error GoTo EH

    '====================================================
    ' 1) Cargar datos "fijos" del empleado desde Empleados / Empleados_Temp
    '====================================================
    encontradoEmp = False

    If CargarEmpleado_Oficial(CStr(numEmp)) Then
        mAltaTemporal = False
        AplicarModoEmpleadoTemporal False
        encontradoEmp = True
    ElseIf CargarEmpleado_Temporal(CStr(numEmp)) Then
        mAltaTemporal = True
        AplicarModoEmpleadoTemporal True
        encontradoEmp = True
    End If
    'OJO: aunque no lo encuentre, igual cargamos incidencias desde BD (puede pasar con bases raras)

    '====================================================
    ' 2) Preparar dict de incidencias por día
    '====================================================
    Set dictCod = CreateObject("Scripting.Dictionary")
    dictCod.CompareMode = vbTextCompare

    '----- hoja BD -----
    Set wsBD = ThisWorkbook.Worksheets("BDIncidencias_Local")
    lastRow = wsBD.Cells(wsBD.Rows.count, "A").End(xlUp).Row

    '----- normalizar tipo periodo (tolerante) -----
    tipoG = UCase$(Trim$(CStr(gTipoPeriodo)))
    If tipoG Like "Q*" Or tipoG Like "QUIN*" Then tipoG = "QUINCENAL"
    If tipoG Like "S*" Or tipoG Like "SEM*" Then tipoG = "SEMANAL"

    '====================================================
    ' 3) Recorrer BD: cargar overrides (si aplica) + incidencias por día
    '====================================================
    For i = 2 To lastRow

        If UCase$(Trim$(CStr(wsBD.Cells(i, "A").value))) <> UCase$(Trim$(CStr(gLoc))) Then GoTo NextI
        If CLng(val(wsBD.Cells(i, "C").value)) <> CLng(numEmp) Then GoTo NextI
        If CLng(val(wsBD.Cells(i, "I").value)) <> CLng(gAnio) Then GoTo NextI
        If CLng(val(wsBD.Cells(i, "J").value)) <> CLng(gMes) Then GoTo NextI

        tipoBD = UCase$(Trim$(CStr(wsBD.Cells(i, "K").value)))
        If tipoBD Like "Q*" Or tipoBD Like "QUIN*" Then tipoBD = "QUINCENAL"
        If tipoBD Like "S*" Or tipoBD Like "SEM*" Then tipoBD = "SEMANAL"
        If tipoBD <> tipoG Then GoTo NextI

        If CLng(val(wsBD.Cells(i, "L").value)) <> CLng(gPeriodo) Then GoTo NextI

        '---- Overrides / extras desde BD (NO pisar si ya hay algo) ----
        If Len(Trim$(CStr(txtAdicional.value))) = 0 Then txtAdicional.value = CStr(wsBD.Cells(i, "P").value)
        If Len(Trim$(CStr(txtObservaciones.value))) = 0 Then txtObservaciones.value = CStr(wsBD.Cells(i, "Q").value)

        If UCase$(Trim$(CStr(gLoc))) = "CAP" Then
            If Len(Trim$(CStr(txtBonoComedor.value))) = 0 Then
                txtBonoComedor.value = CStr(wsBD.Cells(i, 22).value) 'V
            End If
        End If

        '---- incidencias por día ----
        diaBD = CLng(val(wsBD.Cells(i, "M").value))
        cod = Trim$(CStr(wsBD.Cells(i, "O").value))

        If diaBD > 0 Then
            cod = modCatalogoIncidencias.CanonizarCodigo(cod)
            'permitimos vacío: significa sin incidencia
            If Len(cod) > 0 Then dictCod(CStr(diaBD)) = cod
        End If

NextI:
    Next i

    '====================================================
    ' 4) Pintar combos del form según lblDia1..16
    '====================================================
    Dim idx As Long, diaReal As Long
    For idx = 1 To 16
        If Me.Controls("cboDia" & idx).Visible Then

            diaReal = CLng(val(Me.Controls("lblDia" & idx).Caption))

            If dictCod.Exists(CStr(diaReal)) Then
    SetComboByCodigo Me.Controls("cboDia" & idx), dictCod(CStr(diaReal))
Else
    Me.Controls("cboDia" & idx).value = ""
End If


        End If
    Next idx

    'Reaplicar reglas por día (PD/DF/X) después de setear valores
    AplicarReglasDias_EnCombos

    mSilenciarAfterUpdate = False
    Exit Sub

EH:
    mSilenciarAfterUpdate = False
    MsgBox "ERROR en frmIncidencias.CargarDesdeBD: " & err.Number & " - " & err.Description, vbCritical
End Sub

Private Function ComboHasItem(ByVal cbo As MSForms.ComboBox, ByVal val As String) As Boolean
    Dim i As Long
    val = Trim$(CStr(val))
    If Len(val) = 0 Then Exit Function

    For i = 0 To cbo.ListCount - 1
        If StrComp(Trim$(cbo.List(i)), val, vbTextCompare) = 0 Then
            ComboHasItem = True
            Exit Function
        End If
    Next i
End Function

Private Sub SetComboValueSafe(ByVal cbo As MSForms.ComboBox, ByVal val As String, Optional ByVal campo As String = "")
    val = UCase$(Trim$(CStr(val)))

    If Len(val) = 0 Then
        cbo.value = ""
        Exit Sub
    End If

    'NO agregar valores nuevos al catálogo (eso lo vuelve "campo abierto").
    If Not ComboHasItem(cbo, val) Then
        cbo.value = ""
        If Len(campo) > 0 Then
            MsgBox "Valor no existe en catálogo: " & campo & " = '" & val & "'" & vbCrLf & _
                   "Revísalo en CAT_PUESTOS / CAT_ACTIVIDADES (mapeos).", vbExclamation
        End If
        Exit Sub
    End If

    cbo.value = val
End Sub

'===================================================
'  UTIL: agrega únicos de una columna a un ComboBox
'===================================================
Private Sub ComboCargarUnicosDesdeColumna( _
    ByVal cbo As MSForms.ComboBox, _
    ByVal ws As Worksheet, _
    ByVal colIndex As Long, _
    Optional ByVal headerRow As Long = 1, _
    Optional ByVal maxRows As Long = 0)

    Dim dict As Object
    Dim lastRow As Long, r As Long
    Dim v As String

    If colIndex <= 0 Then Exit Sub

    Set dict = CreateObject("Scripting.Dictionary")
    dict.CompareMode = vbTextCompare

    lastRow = ws.Cells(ws.Rows.count, colIndex).End(xlUp).Row
    If maxRows > 0 Then lastRow = Application.Min(lastRow, maxRows)

    For r = headerRow + 1 To lastRow
        v = Trim$(CStr(ws.Cells(r, colIndex).value))
        If Len(v) > 0 Then
            If Not dict.Exists(v) Then dict.Add v, 1
        End If
    Next r

    'Carga en combo
    Dim key As Variant
    For Each key In dict.Keys
        cbo.AddItem CStr(key)
    Next key
End Sub

'Busca fila por No.Empleado aunque venga como texto, número, con espacios, etc.
Private Function FindRowByNumEmp(ByVal ws As Worksheet, ByVal colNum As Long, ByVal numEmp As String) As Long
    Dim lastRow As Long, r As Long
    Dim targetN As Long, v As Variant, vN As Long, vS As String

    targetN = CLng(val(Trim$(numEmp)))
    lastRow = ws.Cells(ws.Rows.count, colNum).End(xlUp).Row

    For r = 2 To lastRow
        v = ws.Cells(r, colNum).value

        'Caso 1: comparación numérica
        vN = CLng(val(v))
        If vN = targetN And targetN <> 0 Then
            FindRowByNumEmp = r
            Exit Function
        End If

        'Caso 2: comparación texto limpia
        vS = Trim$(CStr(v))
        If Len(vS) > 0 Then
            If vS = Trim$(numEmp) Then
                FindRowByNumEmp = r
                Exit Function
            End If
        End If
    Next r

    FindRowByNumEmp = 0
End Function

End Function

        If UCase$(Trim$(lo.ListColumns(j).Name)) = headerName Then
            TableColIndexOne = j
            Exit Function
        End If
    Next j

    TableColIndexOne = 0
End Function

Private Sub LoadComboAnios()
    Dim dict As Object: Set dict = GetYearsFromBDIncidencias()

    Dim yNow As Long: yNow = Year(Date)
    'Siempre incluir actual y 1 atrás
    dict(CStr(yNow)) = yNow
    dict(CStr(yNow - 1)) = yNow - 1

'Quitar futuros (sin borrar mientras iteras)
Dim toRemove As Object: Set toRemove = CreateObject("Scripting.Dictionary")

Dim k As Variant
For Each k In dict.Keys
    If CLng(dict(k)) > yNow Then toRemove(k) = True
Next k

For Each k In toRemove.Keys
    dict.Remove k
Next k


    'Ordenar ascendente
    Dim arr() As Long, n As Long, i As Long, j As Long, tmp As Long
    n = dict.count
    If n = 0 Then Exit Sub

    ReDim arr(1 To n)
    i = 1
    For Each k In dict.Keys
        arr(i) = CLng(dict(k))
        i = i + 1
    Next k

    For i = 1 To n - 1
        For j = i + 1 To n
            If arr(j) < arr(i) Then
                tmp = arr(i): arr(i) = arr(j): arr(j) = tmp
            End If
        Next j
    Next i

    'Cargar combo (de más nuevo a más viejo si prefieres; aquí lo dejo ascendente)
    Me.cboAnio.Clear
    For i = 1 To n
        Me.cboAnio.AddItem CStr(arr(i))
    Next i

    'Default = año actual
    Me.cboAnio.value = CStr(yNow)
End Sub

 '====================================================
'  CAMBIO DE AÑO -> ACTUALIZAR MESES
'====================================================
Private Sub cboAnio_Change()
    Dim anioSel As Long, yNow As Long
    Dim i As Long, maxMes As Long

    If Me.cboAnio.value = "" Then Exit Sub
    If Not IsNumeric(Me.cboAnio.value) Then Exit Sub

    anioSel = CLng(Me.cboAnio.value)
    yNow = Year(Date)

    'Candado: no futuro
    If anioSel > yNow Then
        MsgBox "No se permiten años futuros.", vbExclamation
        Me.cboAnio.value = CStr(yNow)
        Exit Sub
    End If

    'Meses permitidos:
    ' - Año actual: hasta mes actual
    ' - Año pasado: 12 meses
    If anioSel = yNow Then
        maxMes = Month(Date)
    Else
        maxMes = 12
    End If

    Me.cboMes.Clear
    For i = 1 To maxMes
        Me.cboMes.AddItem UCase$(Format$(DateSerial(anioSel, i, 1), "mmmm"))
    Next i

    Me.cboTipoPeriodo.value = ""
    Me.cboPeriodo.Clear
End Sub

Private Sub ActualizarPeriodos()
    Dim anioSel As Long, mesSel As Long
    Dim anioAct As Long, mesAct As Long, diaHoy As Long
    Dim maxPeriodo As Long
    Dim i As Long

    cboPeriodo.Clear
    
    'Si el usuario escribió el mes en vez de seleccionar, forzamos ListIndex correcto
If cboMes.ListIndex < 0 And cboMes.value <> "" Then
    Dim ix As Long
    For ix = 0 To cboMes.ListCount - 1
        If UCase$(Trim$(cboMes.List(ix))) = UCase$(Trim$(cboMes.value)) Then
            cboMes.ListIndex = ix
            Exit For
        End If
    Next ix
End If


    If cboAnio.value = "" Or cboMes.value = "" Or cboTipoPeriodo.value = "" Then Exit Sub
    If cboMes.ListIndex < 0 Then Exit Sub

    anioSel = CLng(cboAnio.value)
    mesSel = cboMes.ListIndex + 1

    anioAct = Year(Date)
    mesAct = Month(Date)
    diaHoy = Day(Date)

    Select Case UCase$(cboTipoPeriodo.value)

        Case "QUINCENAL"
            If (anioSel < anioAct) Or (anioSel = anioAct And mesSel < mesAct) Then
                maxPeriodo = 2
            ElseIf anioSel = anioAct And mesSel = mesAct Then
                If diaHoy <= 15 Then
                    maxPeriodo = 1
                Else
                    maxPeriodo = 2
                End If
            Else
                Exit Sub
            End If

        Case "SEMANAL"
            If (anioSel < anioAct) Or (anioSel = anioAct And mesSel < mesAct) Then
                maxPeriodo = 4
            ElseIf anioSel = anioAct And mesSel = mesAct Then
                Select Case diaHoy
                    Case 1 To 7:   maxPeriodo = 1
                    Case 8 To 14:  maxPeriodo = 2
                    Case 15 To 21: maxPeriodo = 3
                    Case Else:     maxPeriodo = 4
                End Select
            Else
                Exit Sub
            End If

    End Select

    For i = 1 To maxPeriodo
        cboPeriodo.AddItem i
    Next i
End Sub

Private Function NombreHojaMatriz() As String
    Dim tipoCanon As String
    Dim sufTipo As String
    Dim nombre As String
    
    tipoCanon = NormalizarTipoPeriodo(gTipoPeriodo) ' <- CLAVE
    
    If UCase$(tipoCanon) = "SEMANAL" Then
        sufTipo = "S" & gPeriodo
    Else

    End If
    
    'Ejemplo base: M_CUU_2025_12_Q1
    nombre = "M_" & gLoc & "_" & gAnio & "_" & _
             Format$(gMes, "00") & "_" & sufTipo
             
    If Len(nombre) > 31 Then
        nombre = Left$(nombre, 31)
    End If
    
    NombreHojaMatriz = nombre
End Function

Private Function GetHojaMatrizPeriodo() As Worksheet

    nombre = NombreHojaMatriz()
    
    '¿Ya existe la hoja?
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(nombre)
    On Error GoTo 0

        On Error GoTo ErrCrear
        Set ws = ThisWorkbook.Worksheets.Add( _
                    After:=ThisWorkbook.Worksheets(ThisWorkbook.Worksheets.count))
        ws.Name = nombre
        On Error GoTo 0
    End If

    ws.Visible = xlSheetVisible
    
    Set GetHojaMatrizPeriodo = ws
    Exit Function

ErrCrear:

           "Detalle: " & err.Description, vbCritical
    Set GetHojaMatrizPeriodo = Nothing

'====================================================
' 3. Rango (primer y último día) del periodo
'====================================================
Public Sub GetRangoPeriodo( _
        ByVal anio As Long, _
        ByVal mes As Long, _
        ByVal tipoPeriodo As String, _
        ByVal numPeriodo As Long, _
        ByRef diaIni As Long, _
        ByRef diaFin As Long)

    Dim uDia As Long
    uDia = Day(DateSerial(anio, mes + 1, 0))

    Select Case UCase$(tipoPeriodo)

        Case "SEMANAL"
            Select Case numPeriodo
                Case 1: diaIni = 1:  diaFin = 7
                Case 2: diaIni = 8:  diaFin = 14
                Case 3: diaIni = 15: diaFin = 21
                Case 4: diaIni = 22: diaFin = uDia
            End Select

        Case "QUINCENAL"
            Select Case numPeriodo
                Case 1: diaIni = 1:  diaFin = 15
                Case 2: diaIni = 16: diaFin = uDia
            End Select

    End Select

End Sub

'====================================================
' 5. Obtener Adicional, Observaciones y Bono comedor
'    en el periodo actual (primer valor no vacío)
'====================================================
Private Sub ObtenerAdicionalYObs( _
        ByVal numEmp As Long, _
        ByRef adicional As String, _
        ByRef obs As String, _
        Optional ByRef bonoComedor As Variant)

    Dim ws As Worksheet
    Dim ultFila As Long
    Dim i As Long

    adicional = ""
    obs = ""
    bonoComedor = Empty

    Set ws = ThisWorkbook.Worksheets("BDIncidencias_Local")
    ultFila = ws.Cells(ws.Rows.count, "A").End(xlUp).Row

    For i = 2 To ultFila
        If UCase$(Trim$(CStr(ws.Cells(i, "A").value))) = UCase$(Trim$(gLoc)) _
           And CLng(val(ws.Cells(i, "C").value)) = CLng(numEmp) _
           And CLng(val(ws.Cells(i, "I").value)) = CLng(gAnio) _
           And CLng(val(ws.Cells(i, "J").value)) = CLng(gMes) _
           And NormalizarTipoPeriodo(ws.Cells(i, "K").value) = UCase$(gTipoPeriodo) _
           And CLng(val(ws.Cells(i, "L").value)) = CLng(gPeriodo) Then

            If Len(adicional) = 0 Then
                If Len(Trim$(CStr(ws.Cells(i, "P").value))) > 0 Then adicional = CStr(ws.Cells(i, "P").value)
            End If

            If Len(obs) = 0 Then
                If Len(Trim$(CStr(ws.Cells(i, "Q").value))) > 0 Then obs = CStr(ws.Cells(i, "Q").value)
            End If

            ' Bono comedor en columna 22 (V), primer valor no vacío
            If IsEmpty(bonoComedor) Or bonoComedor = "" Then
                If Len(Trim$(CStr(ws.Cells(i, 22).value))) > 0 Then bonoComedor = ws.Cells(i, 22).value
            End If

            If Len(adicional) > 0 And Len(obs) > 0 And Not (IsEmpty(bonoComedor) Or bonoComedor = "") Then
                Exit For
            End If
        End If
    Next i
End Sub

'====================================================
' 6. ¿Ya existen incidencias para este empleado / periodo?
'====================================================
Public Function ExisteIncidenciasEmpleadoPeriodo(ByVal numEmp As Long) As Boolean

    Dim ws As Worksheet
    Dim ultFila As Long
    Dim i As Long

    Set ws = ThisWorkbook.Worksheets("BDIncidencias_Local")
    ultFila = ws.Cells(ws.Rows.count, "A").End(xlUp).Row

    ExisteIncidenciasEmpleadoPeriodo = False
    If ultFila < 2 Then Exit Function

    For i = 2 To ultFila
        If UCase$(Trim$(CStr(ws.Cells(i, 1).value))) = UCase$(Trim$(gLoc)) _
           And CLng(val(ws.Cells(i, 3).value)) = CLng(numEmp) _
           And CLng(val(ws.Cells(i, 9).value)) = CLng(gAnio) _
           And CLng(val(ws.Cells(i, 10).value)) = CLng(gMes) _
           And NormalizarTipoPeriodo(ws.Cells(i, 11).value) = UCase$(gTipoPeriodo) _
           And CLng(val(ws.Cells(i, 12).value)) = CLng(gPeriodo) Then

            ExisteIncidenciasEmpleadoPeriodo = True
            Exit Function
        End If
    Next i
End Function

'====================================================
' 7. Borrar TODAS las incidencias del empleado / periodo
'====================================================
Public Sub BorrarIncidenciasEmpleadoPeriodo(ByVal numEmp As Long)

    Dim ws As Worksheet
    Dim ultFila As Long
    Dim i As Long

    Set ws = ThisWorkbook.Worksheets("BDIncidencias_Local")

    Application.ScreenUpdating = False
    Application.EnableEvents = False

    ' Asegurar que la hoja NO esté protegida
    On Error Resume Next
    ws.Unprotect "AVASA"
    ws.Unprotect "IncidenciasAVASA"
    On Error GoTo 0

    ultFila = ws.Cells(ws.Rows.count, "A").End(xlUp).Row

    'Borrar de abajo hacia arriba
    For i = ultFila To 2 Step -1
        If UCase$(Trim$(CStr(ws.Cells(i, 1).value))) = UCase$(Trim$(gLoc)) _
           And CLng(val(ws.Cells(i, 3).value)) = CLng(numEmp) _
           And CLng(val(ws.Cells(i, 9).value)) = CLng(gAnio) _
           And CLng(val(ws.Cells(i, 10).value)) = CLng(gMes) _
           And NormalizarTipoPeriodo(ws.Cells(i, 11).value) = UCase$(gTipoPeriodo) _
           And CLng(val(ws.Cells(i, 12).value)) = CLng(gPeriodo) Then

            ws.Rows(i).Delete
        End If
    Next i

    ' Re-proteger si aplica
    On Error Resume Next
    ws.Protect Password:="AVASA"
    On Error GoTo 0

    Application.EnableEvents = True
    Application.ScreenUpdating = True
End Sub

'====================================================
' 8. Crear / actualizar un botón (shape) en la hoja
'====================================================
Private Sub CrearOBoton( _
        ByVal ws As Worksheet, _
        ByVal nombre As String, _
        ByVal texto As String, _
        ByVal macro As String, _
        ByVal col As Long)

    Dim sh As Shape
    Dim leftPos As Double, topPos As Double
    Dim ancho As Double, alto As Double

    'Buscar si ya existe el botón
    On Error Resume Next
    Set sh = ws.Shapes(nombre)
    On Error GoTo 0

    'Altura fija de la primera fila
    ws.Rows(1).RowHeight = 37.5

    'Posición y tamaño dentro de la columna indicada
    leftPos = ws.Columns(col).Left + 3
    topPos = ws.Rows(1).Top + 2
    ancho = ws.Columns(col).Width - 6
    alto = ws.Rows(1).Height - 12

    If ancho < 40 Then ancho = 40
    If alto < 18 Then alto = 18

    'Crear o actualizar el shape
    If sh Is Nothing Then
        Set sh = ws.Shapes.AddShape(msoShapeRoundedRectangle, leftPos, topPos, ancho, alto)
        sh.Name = nombre
    Else
        sh.Left = leftPos
        sh.Top = topPos
        sh.Width = ancho
        sh.Height = alto
    End If

    'Formato del botón
    With sh
        'Texto centrado usando TextFrame (no TextFrame2)
        .TextFrame.Characters.Text = texto
        .TextFrame.HorizontalAlignment = xlHAlignCenter
        .TextFrame.VerticalAlignment = xlVAlignCenter

        'Macro asignada
        .OnAction = macro

        'Estilo visual
        .Fill.ForeColor.RGB = RGB(0, 102, 153)
        .line.ForeColor.RGB = RGB(0, 51, 102)
        .TextFrame.Characters.Font.Color = RGB(255, 255, 255)
        .TextFrame.Characters.Font.Size = 10
    End With

End Sub

Private Function HayBDParaPeriodoActual() As Boolean
    Dim wsBD As Worksheet
    Dim lastRow As Long, i As Long
    Dim tipoCanon As String

    tipoCanon = NormalizarTipoPeriodo(gTipoPeriodo)

    On Error Resume Next
    Set wsBD = ThisWorkbook.Worksheets("BDIncidencias_Local")
    On Error GoTo 0

    If wsBD Is Nothing Then
        HayBDParaPeriodoActual = False
        Exit Function
    End If

    lastRow = wsBD.Cells(wsBD.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then
        HayBDParaPeriodoActual = False
        Exit Function
    End If

    For i = 2 To lastRow
        If UCase$(Trim$(CStr(wsBD.Cells(i, "A").value))) = UCase$(Trim$(gLoc)) Then
            ' Año = I, Mes = J, Tipo = K, Periodo = L
            If CLng(val(wsBD.Cells(i, "I").value)) = CLng(gAnio) Then
                If CLng(val(wsBD.Cells(i, "J").value)) = CLng(gMes) Then
                    If NormalizarTipoPeriodo(wsBD.Cells(i, "K").value) = tipoCanon Then
                        If CLng(val(wsBD.Cells(i, "L").value)) = CLng(gPeriodo) Then
                            HayBDParaPeriodoActual = True
                            Exit Function
                        End If
                    End If
                End If
            End If
        End If
    Next i

    HayBDParaPeriodoActual = False
End Function

Public Sub BotonEliminarIncidencia()
    On Error GoTo EH

    Dim ws As Worksheet, fila As Long, numEmp As Long
    Set ws = ActiveSheet

        MsgBox "Esta hoja no parece ser una matriz válida (M_LOC_AAAA_MM_Q#/S#).", vbExclamation

    End If

    modSeguridadIncidencias.ValidarPeriodoAbiertoOrExit

    fila = ActiveCell.Row
    If fila < 3 Then
        MsgBox "Selecciona la fila del empleado que quieres eliminar.", vbExclamation
        Exit Sub
    End If

    If ws.Cells(fila, 3).value = "" Or Not IsNumeric(ws.Cells(fila, 3).value) Then
        MsgBox "La fila seleccionada no contiene un número de empleado válido.", vbExclamation
        Exit Sub
    End If

    numEmp = CLng(val(ws.Cells(fila, 3).value))

    If Not HayBDParaPeriodoActual() Then
        MsgBox "No hay incidencias registradas para este periodo.", vbInformation
        Exit Sub
    End If

        MsgBox "Ese empleado no tiene incidencias registradas en este periodo.", vbInformation

    End If

    BorrarIncidenciasEmpleadoPeriodo numEmp
    
    If Not EmpleadoExisteOficial(numEmp) Then
    BorrarEmpleadoTemporal numEmp
End If


    '2) regenerar matriz (limpieza + rebuild)
    ws.Activate
    GenerarMatrizPeriodoActual

    MsgBox "Listo. Incidencias eliminadas y matriz actualizada.", vbInformation
    Exit Sub

Private Function DiaAsLong(ByVal v As Variant) As Long
    On Error GoTo Fail

    If IsError(v) Or IsEmpty(v) Then GoTo Fail

    If IsDate(v) Then
        DiaAsLong = Day(CDate(v))
        Exit Function
    End If

    If IsNumeric(v) Then
        DiaAsLong = CLng(v)
        Exit Function
    End If

Fail:
    DiaAsLong = 0
End Function

Private Function SetGlobalsDesdeHojaMatriz(ByVal ws As Worksheet) As Boolean
    On Error GoTo Fail

    Dim p() As String, tipoTag As String

    If ws Is Nothing Then GoTo Fail
    If Left$(ws.Name, 2) <> "M_" Then GoTo Fail

    p = Split(ws.Name, "_")
    ' Esperado: M, LOC, YYYY, MM, Q1/S4
    If UBound(p) < 4 Then GoTo Fail

gLoc = Trim$(p(1))
gAnio = CLng(Trim$(p(2)))
gMes = CLng(Trim$(p(3)))

tipoTag = UCase$(Trim$(p(4))) 'Q1 o S4

    If Left$(tipoTag, 1) = "Q" Then
        gTipoPeriodo = "QUINCENAL"
        gPeriodo = CLng(Mid$(tipoTag, 2))
    ElseIf Left$(tipoTag, 1) = "S" Then
        gTipoPeriodo = "SEMANAL"
        gPeriodo = CLng(Mid$(tipoTag, 2))
    Else
        GoTo Fail
    End If

    SetGlobalsDesdeHojaMatriz = True
    Exit Function

Fail:
    SetGlobalsDesdeHojaMatriz = False
End Function

' Devuelve dict(dia)->codigoInc (col M -> col O)
' SOLO para gLoc/gAnio/gMes/gTipoPeriodo/gPeriodo actuales
'====================================================
Public Function GetCodigosPorDia_EmpleadoPeriodo(ByVal numEmp As Long) As Object

    Dim ws As Worksheet, lastRow As Long, i As Long
    Dim d As Long, cod As String
    Dim dict As Object
    Set dict = CreateObject("Scripting.Dictionary")

    Set ws = ThisWorkbook.Worksheets("BDIncidencias_Local")
    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then
        Set GetCodigosPorDia_EmpleadoPeriodo = dict
        Exit Function
    End If

    For i = 2 To lastRow
        If UCase$(CStr(ws.Cells(i, "A").value)) = UCase$(gLoc) _
           And CLng(val(ws.Cells(i, "C").value)) = numEmp _
           And CLng(val(ws.Cells(i, "I").value)) = gAnio _
           And CLng(val(ws.Cells(i, "J").value)) = gMes _
           And NormalizarTipoPeriodo(ws.Cells(i, "K").value) = UCase$(gTipoPeriodo) _
           And CLng(val(ws.Cells(i, "L").value)) = gPeriodo Then

            d = CLng(val(ws.Cells(i, "M").value))
            cod = Trim$(CStr(ws.Cells(i, "O").value))

            If d > 0 And Len(cod) > 0 Then
                dict(CStr(d)) = cod
            End If
        End If
    Next i

    Set GetCodigosPorDia_EmpleadoPeriodo = dict
End Function

Private Sub PintarEncabezadoDias(ByVal ws As Worksheet, ByVal diaIni As Long, ByVal diaFin As Long, ByVal baseColDias As Long)
    Dim d As Long, col As Long
    Dim f As Date

    For d = diaIni To diaFin
        col = baseColDias + (d - diaIni)
        f = DateSerial(gAnio, gMes, d)

        'Domingo (gris)
        If Weekday(f, vbMonday) = 7 Then
            ws.Cells(2, col).Interior.Color = RGB(220, 220, 220)

        'Festivo (rojo clarito)
        ElseIf EsFestivo_Local(f) Then
            ws.Cells(2, col).Interior.Color = RGB(255, 180, 180)
        End If
    Next d
End Sub

    On Error Resume Next
    EsFestivo_Local = modCalendario.EsFestivo(f)
End Function

Public Function NormalizarTipoPeriodo(ByVal v As Variant) As String
    Dim s As String
    s = UCase$(Trim$(CStr(v)))

    If s = "QUINCENAL" Or s = "Q" Or Left$(s, 1) = "Q" Then
        NormalizarTipoPeriodo = "QUINCENAL"
    ElseIf s = "SEMANAL" Or s = "S" Or Left$(s, 1) = "S" Then
        NormalizarTipoPeriodo = "SEMANAL"
    Else
        NormalizarTipoPeriodo = s
    End If
End Function

Private Function NormalizeHeader(ByVal s As String) As String
    s = UCase$(Trim$(CStr(s)))

    ' normalizaciones comunes
    s = Replace(s, ".", "")
    s = Replace(s, ":", "")
    s = Replace(s, "+", "")
    s = Replace(s, "/", "")
    s = Replace(s, "_", "")
    s = Replace(s, vbTab, "")
    
    ' ? CLAVE: quitar TODOS los espacios
    s = Replace(s, " ", "")

    NormalizeHeader = s
End Function

Private Function FindColByHeader(ByVal ws As Worksheet, ByVal headerText As String, Optional ByVal headerRow As Long = 1) As Long
    Dim lastCol As Long, c As Long
    Dim target As String, cur As String

    FindColByHeader = 0
    target = NormalizeHeader(headerText)

    lastCol = ws.Cells(headerRow, ws.Columns.count).End(xlToLeft).Column
    If lastCol <= 0 Then Exit Function

    For c = 1 To lastCol
        cur = NormalizeHeader(ws.Cells(headerRow, c).value)
        If Len(cur) > 0 Then
            If cur = target Or InStr(1, cur, target, vbTextCompare) > 0 Or InStr(1, target, cur, vbTextCompare) > 0 Then
                FindColByHeader = c
                Exit Function
            End If
        End If
    Next c
End Function

Public Sub UpsertEmpleadoTemporal( _
    ByVal grupo As String, _
    ByVal ciudad As String, _
    ByVal numEmp As Long, _
    ByVal nombre As String, _
    ByVal usuarioCars As String, _
    ByVal driverCars As String, _
    ByVal puesto As String, _
    ByVal actividad As String, _
    Optional ByVal periodoAlta As String = "" _
)

    Dim lastRow As Long, r As Long

    Set ws = GetWsEmpleadosTemp(True)
    
ciudad = NormalizarCiudad(ciudad)



    lastRow = ws.Cells(ws.Rows.count, "C").End(xlUp).Row
    If lastRow < 2 Then lastRow = 1

    ' Buscar si ya existe
    For r = 2 To lastRow

' BDIncidencias_Local:
'   X = PuestoOverride
'   Y = ActividadOverride
'====================================================


'----------------------------------------------------
' Lee overrides del periodo (primer valor no vacío)
' Devuelve True si encontró registro del periodo (aunque esté vacío)
'----------------------------------------------------
Public Function GetOverridesEmpleadoPeriodo( _
    ByVal numEmp As Long, _
    ByRef puestoOv As String, _
    ByRef actOv As String) As Boolean

    Dim ws As Worksheet, lastRow As Long, i As Long
    Dim locBD As String

    puestoOv = ""
    actOv = ""
    GetOverridesEmpleadoPeriodo = False

    Set ws = ThisWorkbook.Worksheets("BDIncidencias_Local")
    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then Exit Function

    For i = 2 To lastRow
        locBD = UCase$(Trim$(CStr(ws.Cells(i, "A").value)))

        If locBD = UCase$(Trim$(gLoc)) _
           And CLng(val(ws.Cells(i, "C").value)) = CLng(numEmp) _
           And CLng(val(ws.Cells(i, "I").value)) = CLng(gAnio) _
           And CLng(val(ws.Cells(i, "J").value)) = CLng(gMes) _
           And NormalizarTipoPeriodo(ws.Cells(i, "K").value) = UCase$(gTipoPeriodo) _
           And CLng(val(ws.Cells(i, "L").value)) = CLng(gPeriodo) Then

            GetOverridesEmpleadoPeriodo = True

            If puestoOv = "" Then
                puestoOv = Trim$(CStr(ws.Cells(i, COL_OV_PUESTO).value))
            End If

            If actOv = "" Then
                actOv = Trim$(CStr(ws.Cells(i, COL_OV_ACTIVIDAD).value))
            End If

            If puestoOv <> "" And actOv <> "" Then Exit Function
        End If
    Next i
End Function

'----------------------------------------------------
' Upsert overrides: escribe en la PRIMERA fila encontrada del periodo.
' Si no hay filas del periodo, crea fila "ancla" con día = diaIni.
'----------------------------------------------------
Public Sub UpsertOverridesEmpleadoPeriodo( _
    ByVal numEmp As Long, _
    ByVal puesto As String, _
    ByVal actividad As String)

    Dim ws As Worksheet, lastRow As Long, i As Long
    Dim diaIni As Long, diaFin As Long
    Dim foundRow As Long

    Set ws = ThisWorkbook.Worksheets("BDIncidencias_Local")

    Application.EnableEvents = False
    Application.ScreenUpdating = False

    ' Desproteger por si acaso
    On Error Resume Next
    ws.Unprotect "AVASA"
    ws.Unprotect "IncidenciasAVASA"
    On Error GoTo 0

    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then lastRow = 1

    foundRow = 0

    For i = 2 To lastRow
        If UCase$(Trim$(CStr(ws.Cells(i, "A").value))) = UCase$(Trim$(gLoc)) _
           And CLng(val(ws.Cells(i, "C").value)) = CLng(numEmp) _
           And CLng(val(ws.Cells(i, "I").value)) = CLng(gAnio) _
           And CLng(val(ws.Cells(i, "J").value)) = CLng(gMes) _
           And NormalizarTipoPeriodo(ws.Cells(i, "K").value) = UCase$(gTipoPeriodo) _
           And CLng(val(ws.Cells(i, "L").value)) = CLng(gPeriodo) Then

            foundRow = i
            Exit For
        End If
    Next i

    If foundRow = 0 Then
        'No existe ninguna fila -> crear ancla
        GetRangoPeriodo gAnio, gMes, gTipoPeriodo, gPeriodo, diaIni, diaFin

        foundRow = lastRow + 1
        ws.Cells(foundRow, "A").value = gLoc                  'Loc
        ws.Cells(foundRow, "C").value = CLng(numEmp)          'NumEmp
        ws.Cells(foundRow, "I").value = CLng(gAnio)           'Año
        ws.Cells(foundRow, "J").value = CLng(gMes)            'Mes
        ws.Cells(foundRow, "K").value = UCase$(gTipoPeriodo)  'TipoPeriodo
        ws.Cells(foundRow, "L").value = CLng(gPeriodo)        'Periodo
        ws.Cells(foundRow, "M").value = CLng(diaIni)          'Día (ancla dentro del periodo)
        'O (CodigoInc) lo dejamos vacío
    End If

    ' Escribir overrides (si vienen vacíos, limpiamos)
    ws.Cells(foundRow, COL_OV_PUESTO).value = Trim$(CStr(puesto))
    ws.Cells(foundRow, COL_OV_ACTIVIDAD).value = Trim$(CStr(actividad))

    ' Reproteger
    On Error Resume Next
    ws.Protect Password:="AVASA"
    On Error GoTo 0

    Application.EnableEvents = True
    Application.ScreenUpdating = True
End Sub

'====================================================
' CAT_PUESTOS helpers
' CAT_PUESTOS:
'   A: Puesto_RH (canon)
'   B: EsOperativo (1/0)
'   C: Puesto_Display
'   D: DefaultActividad
'====================================================

Public Function PuestoDisplay(ByVal puestoRH As String) As String
    Dim ws As Worksheet, lastRow As Long, i As Long, p As String

    p = UCase$(Trim$(CStr(puestoRH)))
    PuestoDisplay = Trim$(CStr(puestoRH))

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("CAT_PUESTOS")
    On Error GoTo 0
    If ws Is Nothing Then Exit Function

    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then Exit Function

    For i = 2 To lastRow
        If UCase$(Trim$(CStr(ws.Cells(i, "A").value))) = p Then
            If Len(Trim$(CStr(ws.Cells(i, "C").value))) > 0 Then
                PuestoDisplay = Trim$(CStr(ws.Cells(i, "C").value))
            End If
            Exit Function
        End If
    Next i
End Function

Public Function DefaultActividadDePuesto(ByVal puestoRH As String, Optional ByVal fallback As String = "") As String
    Dim ws As Worksheet, lastRow As Long, i As Long, p As String, defA As String

    p = UCase$(Trim$(CStr(puestoRH)))
    DefaultActividadDePuesto = fallback

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("CAT_PUESTOS")
    On Error GoTo 0
    If ws Is Nothing Then Exit Function

    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then Exit Function

    For i = 2 To lastRow
        If UCase$(Trim$(CStr(ws.Cells(i, "A").value))) = p Then
            defA = Trim$(CStr(ws.Cells(i, "D").value))
            If Len(defA) > 0 Then DefaultActividadDePuesto = defA
            Exit Function
        End If
    Next i
End Function

' CONTEXTO ACTIVO (OFICIAL)
' Lee el NumEmpleado desde la fila activa en la matriz.
' Regla: NumEmpleado está en columna C (3).
'====================================================
Public Function GetNumEmpDesdeSeleccion(Optional ByVal ws As Worksheet) As Long
    On Error GoTo salir

    If ws Is Nothing Then Set ws = ActiveSheet

    Dim fila As Long
    fila = ActiveCell.Row

    ' Evitar header / zonas no válidas (ajusta si tu header es otra fila)
    If fila < 3 Then GoTo salir

    Dim v As Variant
    v = ws.Cells(fila, 3).value ' Columna C

    If Len(Trim(CStr(v))) = 0 Then GoTo salir
    If Not IsNumeric(v) Then GoTo salir

    GetNumEmpDesdeSeleccion = CLng(v)
    Exit Function

salir:
    GetNumEmpDesdeSeleccion = 0
End Function

' BOTÓN ÚNICO "OPCIONES" (router)
'====================================================
Public Sub BotonOpciones()

    ' 1) Validar que estás en hoja matriz + setear globals
    If Not SetGlobalsDesdeHojaMatriz(ActiveSheet) Then
        MsgBox "Esta acción solo se puede usar desde una matriz de incidencias.", vbExclamation
        Exit Sub
    End If

    ' 2) Obtener NumEmpleado desde selección (col C)
    Dim numEmp As Long
    numEmp = GetNumEmpDesdeSeleccion(ActiveSheet)

    If numEmp = 0 Then
        MsgBox "Selecciona una fila válida de un empleado (NumEmpleado en columna C).", vbExclamation
        Exit Sub
    End If
gCtxWsName = ActiveSheet.Name
gCtxFila = ActiveCell.Row
gCtxNumEmp = numEmp

    ' 3) Armar textos
    Dim nom As String
    nom = GetNombreEmpleadoDesdeSeleccion(ActiveSheet)

    Dim sEmpleado As String
    If Len(nom) > 0 Then
        sEmpleado = CStr(numEmp) & " - " & nom
    Else
        sEmpleado = CStr(numEmp)
    End If

    Dim sPeriodo As String
    sPeriodo = gLoc & " | " & gAnio & "-" & Format(gMes, "00") & " | " & gTipoPeriodo & gPeriodo

    ' 4) Mostrar frmOpciones + setear labels (sin tronar si no existen)
    On Error Resume Next
    SafeSetCaption frmOpciones, "lblEmpleado", sEmpleado
    SafeSetCaption frmOpciones, "lblPeriodo", sPeriodo
    On Error GoTo 0

    frmOpciones.Show

End Sub

'====================================================
' PUBLIC: Obtener Adicional / Observaciones / Bono (primer valor no vacío)
' para el periodo actual (gLoc/gAnio/gMes/gTipoPeriodo/gPeriodo)
'====================================================
Public Sub GetAdicionalObsBono_EmpleadoPeriodo( _
        ByVal numEmp As Long, _
        ByRef adicional As String, _
        ByRef obs As String, _
        Optional ByRef bonoComedor As Variant)

    Dim ws As Worksheet
    Dim ultFila As Long, i As Long

    adicional = ""
    obs = ""
    bonoComedor = Empty

    Set ws = ThisWorkbook.Worksheets("BDIncidencias_Local")
    ultFila = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If ultFila < 2 Then Exit Sub

    For i = 2 To ultFila
        If UCase$(Trim$(CStr(ws.Cells(i, "A").value))) = UCase$(Trim$(gLoc)) _
           And CLng(val(ws.Cells(i, "C").value)) = CLng(numEmp) _
           And CLng(val(ws.Cells(i, "I").value)) = CLng(gAnio) _
           And CLng(val(ws.Cells(i, "J").value)) = CLng(gMes) _
           And NormalizarTipoPeriodo(ws.Cells(i, "K").value) = UCase$(gTipoPeriodo) _
           And CLng(val(ws.Cells(i, "L").value)) = CLng(gPeriodo) Then

            If Len(adicional) = 0 Then
                If Len(Trim$(CStr(ws.Cells(i, "P").value))) > 0 Then adicional = CStr(ws.Cells(i, "P").value)
            End If

            If Len(obs) = 0 Then
                If Len(Trim$(CStr(ws.Cells(i, "Q").value))) > 0 Then obs = CStr(ws.Cells(i, "Q").value)
            End If

            If IsEmpty(bonoComedor) Or bonoComedor = "" Then
                If Len(Trim$(CStr(ws.Cells(i, 22).value))) > 0 Then bonoComedor = ws.Cells(i, 22).value
            End If

            If Len(adicional) > 0 And Len(obs) > 0 And Not (IsEmpty(bonoComedor) Or bonoComedor = "") Then Exit For
        End If
    Next i
End Sub

'====================================================
' Cerrar periodo actual (manual)
' - Set Status = "CERRADO" en tblPeriodos
' - Regenera matriz
' - Aplica cierre/protección en la hoja matriz
'====================================================
Public Sub CerrarPeriodoActual()
    On Error GoTo EH

    '--- Validar contexto ---
    If gLoc = "" Or gAnio = 0 Or gMes = 0 Or gTipoPeriodo = "" Or gPeriodo = 0 Then
        MsgBox "Falta contexto de periodo (loc/año/mes/tipo/periodo). Entra por menú o abre la matriz válida.", vbExclamation
        Exit Sub
    End If

    '--- Confirmación ---
    Dim msg As String
    msg = "Vas a CERRAR el periodo actual:" & vbCrLf & vbCrLf & _
          "Loc: " & gLoc & vbCrLf & _
          "Periodo: " & gAnio & "-" & Format$(gMes, "00") & " " & UCase$(gTipoPeriodo) & " " & gPeriodo & vbCrLf & vbCrLf & _
          "Esto dejará el periodo en SOLO LECTURA (no se podrá guardar/editar)." & vbCrLf & _
          "¿Confirmas cerrar el periodo?"
    If MsgBox(msg, vbExclamation + vbYesNo + vbDefaultButton2, "Cerrar periodo") <> vbYes Then
        Exit Sub
    End If

    '--- Escribir status en tblPeriodos ---
    modPeriodos.SetPeriodoStatus gLoc, gAnio, gMes, gTipoPeriodo, gPeriodo, "CERRADO"

    '--- Regenerar y aplicar cierre visual/protección ---
    Dim wsMat As Worksheet
    Set wsMat = GetHojaMatrizPeriodo() 'usa tu getter actual
    If wsMat Is Nothing Then
        'si no encontró la hoja, de todos modos ya quedó cerrado en tblPeriodos
        MsgBox "Periodo cerrado, pero no encontré la hoja de matriz para refrescar.", vbInformation
        Exit Sub
    End If

    'Regenerar matriz (esto al final vuelve a llamar AplicarCierreEnMatriz)
    GenerarMatrizPeriodoActual

    'Por si acaso (doble seguro)
    modSeguridadIncidencias.AplicarCierreEnMatriz wsMat

    MsgBox "Periodo CERRADO. El sistema quedó en modo solo lectura para este periodo.", vbInformation
    Exit Sub

EH:
    MsgBox "ERROR en CerrarPeriodoActual: " & err.Number & " - " & err.Description, vbCritical
End Sub


'====================================================
' COMPLETAR PERIODO (AUTO)
' - Recorre SOLO empleados visibles en la matriz (fila 3..)
' - NO pisa capturas (ni matriz ni BD)
' - Escribe en BDIncidencias_Local (no solo en la matriz)
' - Evita completado si el empleado está marcado como ELIMINADO (si existe la hoja)
'====================================================

Private Function PeriodoKeyActual() As String
    PeriodoKeyActual = UCase$(Trim$(gLoc)) & "|" & gAnio & "|" & Format$(gMes, "00") & "|" & _
                       UCase$(NormalizarTipoPeriodo(gTipoPeriodo)) & "|" & gPeriodo
End Function

'Si aún no tienes la hoja, esto regresa False y no estorba.
Private Function EsEmpleadoEliminadoPeriodo(ByVal numEmp As Long) As Boolean
    On Error GoTo salir

    Dim ws As Worksheet, lastRow As Long, i As Long
    Dim key As String
    key = PeriodoKeyActual()

    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("Empleados_Eliminados")
    On Error GoTo 0
    If ws Is Nothing Then GoTo salir

    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then GoTo salir

    For i = 2 To lastRow
        If UCase$(Trim$(CStr(ws.Cells(i, "A").value))) = UCase$(Trim$(gLoc)) _
           And UCase$(Trim$(CStr(ws.Cells(i, "B").value))) = key _
           And CLng(val(ws.Cells(i, "C").value)) = CLng(numEmp) Then
            EsEmpleadoEliminadoPeriodo = True
            Exit Function
        End If
    Next i

salir:
    EsEmpleadoEliminadoPeriodo = False
End Function


Public Sub CompletarPeriodoActual()
    On Error GoTo EH

    'Debe estar en una matriz válida
    If Not SetGlobalsDesdeHojaMatriz(ActiveSheet) Then
        MsgBox "Esta acción solo se puede usar desde una matriz (M_LOC_AAAA_MM_Q#/S#).", vbExclamation
        Exit Sub
    End If

    'No permitir si ya está cerrado
    modSeguridadIncidencias.ValidarPeriodoAbiertoOrExit

    Dim wsMat As Worksheet, wsBD As Worksheet
    Set wsMat = ActiveSheet
    Set wsBD = ThisWorkbook.Worksheets("BDIncidencias_Local")

    Dim diaIni As Long, diaFin As Long, numDias As Long
    GetRangoPeriodo gAnio, gMes, gTipoPeriodo, gPeriodo, diaIni, diaFin
    numDias = diaFin - diaIni + 1
'----------------------------------------
' CORTE: no autocompletar días posteriores a HOY
' Regla segura: completar solo hasta AYER
'----------------------------------------
Dim fIni As Date, fFin As Date, fCorte As Date
Dim diaFinAuto As Long

fIni = DateSerial(gAnio, gMes, diaIni)
fFin = DateSerial(gAnio, gMes, diaFin)
fCorte = Date - 1 ' <-- AYER

'Si el periodo todavía no empieza (respecto a ayer), no completar nada
If fCorte < fIni Then
    MsgBox "Aún no hay días para autocompletar en este periodo (corte: " & Format$(fCorte, "dd/mm/yyyy") & ").", vbInformation
    Exit Sub
End If

'Si ya pasó todo el periodo, completar completo
If fCorte >= fFin Then
    diaFinAuto = diaFin
Else
    'Corte dentro del mismo mes del periodo (tu diseño es por mes)
    diaFinAuto = Day(fCorte)
    If diaFinAuto < diaIni Then diaFinAuto = diaIni - 1
End If

If diaFinAuto < diaIni Then
    MsgBox "Con el corte a AYER, no hay días dentro del rango a completar.", vbInformation
    Exit Sub
End If

    Dim baseColDias As Long
    baseColDias = 9 'I, igual que tu GenerarMatriz

    'Confirmación
    Dim msg As String
    msg = "Vas a COMPLETAR incidencias faltantes (AUTO) para:" & vbCrLf & _
          gLoc & " | " & gAnio & "-" & Format$(gMes, "00") & " " & UCase$(gTipoPeriodo) & " " & gPeriodo & vbCrLf & vbCrLf & _
          "Solo llena celdas vacías. NO pisa capturas existentes." & vbCrLf & _
          "Solo opera sobre empleados visibles en la matriz." & vbCrLf & _
          "¿Continuar?"
    If MsgBox(msg, vbQuestion + vbYesNo, "Completar incidencias") <> vbYes Then Exit Sub

    Application.ScreenUpdating = False
    Application.EnableEvents = False

    'Desproteger BD por si acaso
    On Error Resume Next
    wsBD.Unprotect "AVASA"
    wsBD.Unprotect "IncidenciasAVASA"
    On Error GoTo EH

    'Index rápido de lo que ya existe en BD para este periodo: key emp|dia
    Dim dictExiste As Object
    Set dictExiste = CreateObject("Scripting.Dictionary")

    Dim lastBD As Long, i As Long
    lastBD = wsBD.Cells(wsBD.Rows.count, "A").End(xlUp).Row

    Dim empBD As Long, diaBD As Long
    For i = 2 To lastBD
        If UCase$(Trim$(CStr(wsBD.Cells(i, "A").value))) = UCase$(Trim$(gLoc)) _
           And CLng(val(wsBD.Cells(i, "I").value)) = gAnio _
           And CLng(val(wsBD.Cells(i, "J").value)) = gMes _
           And NormalizarTipoPeriodo(wsBD.Cells(i, "K").value) = UCase$(gTipoPeriodo) _
           And CLng(val(wsBD.Cells(i, "L").value)) = gPeriodo Then

            empBD = CLng(val(wsBD.Cells(i, "C").value))
            diaBD = CLng(val(wsBD.Cells(i, "M").value))
            If empBD > 0 And diaBD >= diaIni And diaBD <= diaFin Then
                dictExiste(CStr(empBD) & "|" & CStr(diaBD)) = True
            End If
        End If
    Next i

    'Recorrer empleados en matriz (desde fila 3)
    Dim fila As Long, lastMat As Long, numEmp As Long
    lastMat = wsMat.Cells(wsMat.Rows.count, 3).End(xlUp).Row

    Dim d As Long, colDia As Long, f As Date
    Dim codAuto As String, key As String
    Dim ciudad As String, usuarioCars As String, driverCars As String
    Dim puesto As String, actividad As String, nombre As String
    Dim newRow As Long

    For fila = 3 To lastMat

        If Not IsNumeric(wsMat.Cells(fila, 3).value) Then GoTo NextFila
        numEmp = CLng(val(wsMat.Cells(fila, 3).value))
        If numEmp <= 0 Then GoTo NextFila

        'Evitar eliminados del periodo (si está implementada la hoja)
        If EsEmpleadoEliminadoPeriodo(numEmp) Then GoTo NextFila

        'datos “base” desde la matriz (col B..H)
        ciudad = CStr(wsMat.Cells(fila, 2).value)
        usuarioCars = CStr(wsMat.Cells(fila, 4).value)
        driverCars = CStr(wsMat.Cells(fila, 5).value)
        puesto = CStr(wsMat.Cells(fila, 6).value)
        actividad = CStr(wsMat.Cells(fila, 7).value)
        nombre = CStr(wsMat.Cells(fila, 8).value)

        For d = diaIni To diaFinAuto
            colDia = baseColDias + (d - diaIni)

            'Si ya hay algo en matriz, no completar
            If Len(Trim$(CStr(wsMat.Cells(fila, colDia).value))) > 0 Then GoTo NextDia

            key = CStr(numEmp) & "|" & CStr(d)

            'Si ya existe en BD, no completar
            If dictExiste.Exists(key) Then GoTo NextDia

            f = DateSerial(gAnio, gMes, d)

            'Reglas:
            'Domingo => PD
            'Festivo => DF
            'Normal => X
            If Weekday(f, vbMonday) = 7 Then
                codAuto = "PD"
            ElseIf EsFestivo_Local(f) Then
                codAuto = "DF"
            Else
                codAuto = "X"
            End If

            'Canonizar por si acaso
            codAuto = modCatalogoIncidencias.CanonizarCodigo(codAuto)

            'Append en BD
            newRow = wsBD.Cells(wsBD.Rows.count, "A").End(xlUp).Row + 1

            With wsBD
                .Cells(newRow, 1).value = gLoc
                .Cells(newRow, 2).value = ciudad
                .Cells(newRow, 3).value = numEmp
                .Cells(newRow, 4).value = usuarioCars
                .Cells(newRow, 5).value = driverCars
                .Cells(newRow, 6).value = puesto
                .Cells(newRow, 7).value = actividad
                .Cells(newRow, 8).value = nombre

                .Cells(newRow, 9).value = gAnio
                .Cells(newRow, 10).value = gMes
                .Cells(newRow, 11).value = gTipoPeriodo
                .Cells(newRow, 12).value = gPeriodo
                .Cells(newRow, 13).value = d
                .Cells(newRow, 14).value = f
                .Cells(newRow, 14).NumberFormat = "dd/mm/yyyy"

                .Cells(newRow, 15).value = codAuto
                .Cells(newRow, 16).value = "" 'Adicional
                .Cells(newRow, 17).value = "" 'Observaciones
                .Cells(newRow, 18).value = Environ$("username")
                .Cells(newRow, 19).value = Now
                .Cells(newRow, 20).value = "AUTO"

                If UCase$(gLoc) = "CAP" Then .Cells(newRow, 22).value = "" 'bono
            End With

            dictExiste(key) = True

NextDia:
        Next d

NextFila:
    Next fila

    'Reproteger BD
    On Error Resume Next
    wsBD.Protect Password:="AVASA"
    On Error GoTo EH

    Application.EnableEvents = True
    Application.ScreenUpdating = True

    'Refrescar matriz
    GenerarMatrizPeriodoActual

    MsgBox "Listo: incidencias faltantes completadas en BD (AUTO) y matriz regenerada.", vbInformation
    Exit Sub

EH:
    On Error Resume Next
    Application.EnableEvents = True
    Application.ScreenUpdating = True
    MsgBox "ERROR en CompletarPeriodoActual: " & err.Number & " - " & err.Description, vbCritical
End Sub

Public Function NormalizarCiudad(ByVal ciudadRaw As String) As String
    Dim s As String, p As Long
    s = UCase$(Trim$(CStr(ciudadRaw)))

    'Si viene como "LOC - CIUDAD", toma lo de la derecha (genérico para ACA, CAP, MZT, etc.)
    p = InStr(1, s, " - ", vbTextCompare)
    If p > 0 Then
        s = Mid$(s, p + 3)
    End If

    'limpieza extra
    s = Replace(s, "  ", " ")
    s = Trim$(s)

    NormalizarCiudad = s
End Function

        ws.Name = "ETL_Uni_PuestoActividad"
    End If

    ws.Range("A1").value = "PUESTO"
    ws.Range("B1").value = "ACTIVIDAD"

    If loDB.ListRows.count > 0 Then

            v = Trim$(CStr(loDB.DataBodyRange.Cells(i, idxPuesto).value))

            v = Trim$(CStr(loDB.DataBodyRange.Cells(i, idxAct).value))
            If Len(v) > 0 Then If Not dictA.Exists(v) Then dictA.Add v, 1
        Next i
    End If

    r = 2
    Dim k As Variant
    For Each k In dictP.Keys
        ws.Cells(r, "A").value = CStr(k)
        r = r + 1
    Next k

Public Function NormalizarTipoNomina(ByVal v As Variant) As String
    Dim s As String
    s = UCase$(Trim$(CStr(v)))

    'limpiezas comunes
    s = Replace(s, ".", "")
    s = Replace(s, "-", "")
    s = Replace(s, "_", "")
    s = WorksheetFunction.Trim(s)

    Select Case s
        Case "SEM", "SEMAN", "SEMANAL", "S"
            NormalizarTipoNomina = "SEMANAL"
        Case "QUIN", "QUINC", "QUINCENAL", "Q"
            NormalizarTipoNomina = "QUINCENAL"
        Case Else
            NormalizarTipoNomina = ""   'desconocido -> falla filtro estricto
    End Select
End Function

Public Function NormalizarLoc(ByVal s As String) As String
    Dim t As String, p As Long
    t = UCase$(Trim$(CStr(s)))

    ' si viene "MTA - MERIDA" -> "MTA"
    p = InStr(1, t, " - ", vbTextCompare)
    If p > 0 Then t = Left$(t, p - 1)

    NormalizarLoc = Trim$(t)
End Function

'==========================
'  CIERRE AUTOMÁTICO
'==========================
'Se considera "cerrado" cuando:
'   Now >= FechaFinPeriodo + (LockWindowHours / 24)
'LockWindowHours viene de Config (ej 48)
'
'NEW:
' - Si tblPeriodos.Status = CERRADO -> bloquea siempre.
' - Si tblPeriodos.LockWindowHoursOverride tiene valor -> usa ese valor en vez del default.

Public Function GetLockWindowHours_Default() As Double
    'Default: 48 horas
    Dim v As Variant
    v = GetConfig("LockWindowHours", 48)

    If IsNumeric(v) Then
        GetLockWindowHours_Default = CDbl(v)
    Else
        GetLockWindowHours_Default = 48#
    End If
End Function

Public Function GetLockWindowHours_Efectivo() As Double
    'Devuelve override si existe; si no, default de Config
    On Error GoTo Fail

    Dim vOvr As Variant
    vOvr = modPeriodos.GetLockOverrideHours(gLoc, gAnio, gMes, gTipoPeriodo, gPeriodo)

    If IsEmpty(vOvr) Then
        GetLockWindowHours_Efectivo = GetLockWindowHours_Default()
    ElseIf IsNumeric(vOvr) Then
        GetLockWindowHours_Efectivo = CDbl(vOvr)
    Else
        GetLockWindowHours_Efectivo = GetLockWindowHours_Default()
    End If
    Exit Function

Fail:
    GetLockWindowHours_Efectivo = GetLockWindowHours_Default()
End Function

Public Function GetFechaFinPeriodoActual() As Date
    'Calcula fecha fin del periodo actual usando globals
    'Requiere: gAnio, gMes, gTipoPeriodo, gPeriodo (modGlobal)
    Dim diaIni As Long, diaFin As Long
    diaIni = 0: diaFin = 0

    Call ObtenerRangoPeriodo_Local(gAnio, gMes, gTipoPeriodo, gPeriodo, diaIni, diaFin)
    GetFechaFinPeriodoActual = DateSerial(gAnio, gMes, diaFin)
End Function

Public Function GetFechaCierrePeriodoActual() As Date
    GetFechaCierrePeriodoActual = GetFechaFinPeriodoActual() + (GetLockWindowHours_Efectivo() / 24#)
End Function

Public Function PeriodoActualEstaCerrado(Optional ByRef cierre As Date) As Boolean
    cierre = GetFechaCierrePeriodoActual()
    PeriodoActualEstaCerrado = (Now >= cierre)
End Function

Public Function PermiteEdicionPeriodo(Optional ByRef msgBloqueo As String) As Boolean
    On Error GoTo Fail

    '-----------------------------------------
    ' 1) OVERRIDE MANUAL: Status=CERRADO
    '-----------------------------------------
    Dim st As String
    st = modPeriodos.GetPeriodoStatus(gLoc, gAnio, gMes, gTipoPeriodo, gPeriodo)

    If st = "CERRADO" Then
        msgBloqueo = "Periodo CERRADO (manual)." & vbCrLf & _
                     "Si necesitas reabrir, solicítalo a Nómina."
        PermiteEdicionPeriodo = False
        Exit Function
    End If

'-----------------------------------------
' 2) CIERRE AUTOMÁTICO: fin + horas (override o default)
'-----------------------------------------
Dim cierre As Date
If PeriodoActualEstaCerrado(cierre) Then

    'NEW: dejar huella del cierre automático
    On Error Resume Next
    modPeriodos.SetPeriodoStatus gLoc, gAnio, gMes, gTipoPeriodo, gPeriodo, "CERRADO"
    On Error GoTo 0

    msgBloqueo = "Periodo CERRADO." & vbCrLf & _
                 "Cierre automático: " & Format$(cierre, "dd/mm/yyyy hh:mm") & vbCrLf & _
                 "Si necesitas reabrir, solicítalo a Nómina."
    PermiteEdicionPeriodo = False
Else
    PermiteEdicionPeriodo = True
End If
Exit Function


Fail:
    'Si algo falla, no bloqueamos por error: usamos la lógica actual (por fecha + default)
    Dim cierre2 As Date
    cierre2 = GetFechaFinPeriodoActual() + (GetLockWindowHours_Default() / 24#)
    If Now >= cierre2 Then
        msgBloqueo = "Periodo CERRADO." & vbCrLf & _
                     "Cierre automático: " & Format$(cierre2, "dd/mm/yyyy hh:mm") & vbCrLf & _
                     "Si necesitas reabrir, solicítalo a Nómina."
        PermiteEdicionPeriodo = False
    Else
        PermiteEdicionPeriodo = True
    End If
End Function

Public Sub ValidarPeriodoAbiertoOrExit()
    Dim m As String
    If Not PermiteEdicionPeriodo(m) Then
        MsgBox m, vbExclamation, "Bloqueo por cierre"
        err.Raise vbObjectError + 513, "modSeguridadIncidencias", "Periodo cerrado"
    End If
End Sub

'-----------------------------------------
' Helper local: rango periodo
'-----------------------------------------
Private Sub ObtenerRangoPeriodo_Local( _
        ByVal anio As Long, _
        ByVal mes As Long, _
        ByVal tipoPeriodo As String, _
        ByVal numPeriodo As Long, _
        ByRef diaIni As Long, _
        ByRef diaFin As Long)

    Dim uDia As Long
    uDia = Day(DateSerial(anio, mes + 1, 0))

    Select Case UCase$(tipoPeriodo)
        Case "SEMANAL"
            Select Case numPeriodo
                Case 1: diaIni = 1:  diaFin = 7
                Case 2: diaIni = 8:  diaFin = 14
                Case 3: diaIni = 15: diaFin = 21
                Case 4: diaIni = 22: diaFin = uDia
            End Select

        Case "QUINCENAL"
            Select Case numPeriodo
                Case 1: diaIni = 1:  diaFin = 15
                Case 2: diaIni = 16: diaFin = uDia
            End Select
    End Select
End Sub


'-----------------------------------------
' Public wrapper: rango periodo (para otros módulos)
'-----------------------------------------
Public Sub ObtenerRangoPeriodo_Public( _
        ByVal anio As Long, _
        ByVal mes As Long, _
        ByVal tipoPeriodo As String, _
        ByVal numPeriodo As Long, _
        ByRef diaIni As Long, _
        ByRef diaFin As Long)

    Call ObtenerRangoPeriodo_Local(anio, mes, tipoPeriodo, numPeriodo, diaIni, diaFin)

End Sub

'Busca un header exacto en las columnas de la tabla (case-insensitive)
Private Function TableColIndexOne(ByVal lo As ListObject, ByVal headerName As String) As Long

        If UCase$(Trim$(lo.ListColumns(j).Name)) = headerName Then
            TableColIndexOne = j
            Exit Function
        End If
    Next j

    TableColIndexOne = 0
End Function

Option Explicit

Public Sub BuscarMatrizPeriodoAdmin()

    Dim codigo As String
    Dim ws As Worksheet
    Dim hojaEsperada As Worksheet
    Dim lista() As String
    Dim n As Long, i As Long
    Dim msg As String
    Dim resp As Variant
    Dim nombreEsperado As String
    
    '------------------------------------
    ' Pedir periodo: 2025_12_Q1, 2026_03_S2, etc.
    '------------------------------------
    codigo = InputBox( _
        Prompt:="Pega el periodo con el formato:" & vbCrLf & _
                "AAAA_MM_Q#  o  AAAA_MM_S#" & vbCrLf & _
                "Ejemplo:  2025_12_Q1", _
        Title:="Buscar matriz de incidencias")
    
    codigo = Trim(UCase$(codigo))
    codigo = Replace(codigo, " ", "")
    
    If codigo = "" Then Exit Sub
    
    ' Validación muy ligera: 2 guiones bajos
    If UBound(Split(codigo, "_")) <> 2 Then
        MsgBox "El texto no parece un periodo válido (ej. 2025_12_Q1).", vbExclamation
        Exit Sub
    End If
    
    '------------------------------------
    ' 1) Intento directo con el patrón "M_LOC_periodo"
    '    (si el archivo está bien hecho, cada archivo solo tiene 1 LOC)
    '------------------------------------
    On Error Resume Next
    If gLoc <> "" Then
        nombreEsperado = "M_" & gLoc & "_" & codigo
        Set hojaEsperada = ThisWorkbook.Worksheets(nombreEsperado)
    End If
    On Error GoTo 0
    
    If Not hojaEsperada Is Nothing Then
        hojaEsperada.Visible = xlSheetVisible
        hojaEsperada.Activate
        Exit Sub
    End If
    
    '------------------------------------
    ' 2) Plan B: buscar cualquier hoja que termine en "_periodo"
    '    Ej: * _2025_12_Q1
    '------------------------------------
    n = 0
    For Each ws In ThisWorkbook.Worksheets
        If UCase$(ws.Name) <> "MENU" Then
            If ws.Name Like "*_" & codigo Then
                n = n + 1
                ReDim Preserve lista(1 To n)
                lista(n) = ws.Name
            End If
        End If
    Next ws
    
    Select Case n
        Case 0
            MsgBox "No se encontró ninguna hoja cuyo nombre termine en '" & codigo & "'.", _
                   vbInformation
            Exit Sub
        
        Case 1
            Set hojaEsperada = ThisWorkbook.Worksheets(lista(1))
            hojaEsperada.Visible = xlSheetVisible
            hojaEsperada.Activate
        
        Case Else
            ' Hay varias coincidencias, que elijas una
            msg = "Se encontraron varias hojas con el periodo " & codigo & ":" & vbCrLf & vbCrLf
            For i = 1 To n
                msg = msg & i & ") " & lista(i) & vbCrLf
            Next i
            msg = msg & vbCrLf & "Escribe el número de la hoja a la que quieres ir:"
            
            resp = InputBox(msg, "Seleccionar hoja", "1")
            If resp = "" Then Exit Sub
            If Not IsNumeric(resp) Then
                MsgBox "Valor no válido.", vbExclamation
                Exit Sub
            End If
            i = CLng(resp)
            If i < 1 Or i > n Then
                MsgBox "El número no está en la lista.", vbExclamation
                Exit Sub
            End If
            
            Set hojaEsperada = ThisWorkbook.Worksheets(lista(i))
            hojaEsperada.Visible = xlSheetVisible
            hojaEsperada.Activate
    End Select

End Sub

'==========================

'==========================

Public Function GetConfig(ByVal cfgKey As String, Optional ByVal defaultValue As String = "") As String
    Dim lo As ListObject, r As ListRow
    Dim v As Variant

    On Error GoTo SAFEEXIT

    Set lo = ThisWorkbook.Worksheets("Config").ListObjects("tblConfig")

    For Each r In lo.ListRows
        If UCase$(Trim$(CStr(r.Range.Cells(1, 1).value))) = UCase$(Trim$(cfgKey)) Then

            v = r.Range.Cells(1, 2).value

            If IsError(v) Or IsNull(v) Then
                GetConfig = defaultValue
            Else
                GetConfig = Trim$(CStr(v))
                If Len(GetConfig) = 0 Then GetConfig = defaultValue
            End If

            Exit Function
        End If
    Next r

SAFEEXIT:
    If Len(GetConfig) = 0 Then GetConfig = defaultValue
End Function

    Dim lo As ListObject, r As ListRow
    Set lo = ThisWorkbook.Worksheets("Config").ListObjects("tblConfig")

    For Each r In lo.ListRows
        If UCase$(Trim$(CStr(r.Range.Cells(1, 1).value))) = UCase$(Trim$(cfgKey)) Then
            r.Range.Cells(1, 2).value = cfgValue
            Exit Sub
        End If
    Next r

    Set r = lo.ListRows.Add
    r.Range.Cells(1, 1).value = cfgKey
    r.Range.Cells(1, 2).value = cfgValue
End Sub

'============================================================
' modGeneradorLocaciones (SIN MASTER EXTERNO)
' Genera archivos por locación + INYECCIÓN de Empleados desde el MISMO TEMPLATE
' Source:  Empleados!tblEmpleados
' Dest:    Empleados!tblEmpleados_Local
'============================================================

'--- Catálogo locaciones (en ESTE archivo maestro)
Private Const SHEET_LOC As String = "Locaciones"
Private Const TBL_LOC As String = "tblLocaciones"
Private Const COL_ACTIVE As String = "Active"
Private Const COL_LOCCODE As String = "LocationCode"

'--- Empleados (en el template / archivo hijo)
Private Const EMP_SHEET As String = "Empleados"
Private Const EMP_TBL As String = "tblEmpleados"
Private Const EMP_COL_LOCCODE As String = "LocCode"


Public Sub GenerarArchivosPorLocacion()
    On Error GoTo EH

    Dim stepName As String: stepName = "INIT"
    Dim logText As String: logText = ""

    If Len(ThisWorkbook.path) = 0 Then
        MsgBox "El archivo maestro no está guardado. Guarda el .xlsm en disco (Desktop) y vuelve a intentar.", vbCritical
        Exit Sub
    End If

    Dim basePath As String
    basePath = CleanPath(GetConfigValue("GerentesRootPath", ""))

    If Len(basePath) = 0 Then err.Raise vbObjectError + 1, , "Config: falta GerentesRootPath"
    If Right$(basePath, 1) <> "\" Then basePath = basePath & "\"

LogMem logText, "=== INICIO GENERACIÓN ==="
LogMem logText, "Root: " & basePath
LogMem logText, "Empleados table: " & EMP_SHEET & "!tblEmpleados (filtrado por LocCode)"


    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False

    'Locaciones
    stepName = "LOC: Read tblLocaciones"
    Dim loLoc As ListObject
    Set loLoc = ThisWorkbook.Worksheets(SHEET_LOC).ListObjects(TBL_LOC)

    Dim idxActive As Long, idxCode As Long
    idxActive = loLoc.ListColumns(COL_ACTIVE).Index
    idxCode = loLoc.ListColumns(COL_LOCCODE).Index

    Dim r As ListRow
    Dim locCode As String, folderOut As String, fullPath As String, tmpPath As String
    Dim wbNew As Workbook
    Dim okCount As Long, errCount As Long
    okCount = 0: errCount = 0

    For Each r In loLoc.ListRows
        If val(r.Range.Cells(1, idxActive).value) = 1 Then

            locCode = Trim$(CStr(r.Range.Cells(1, idxCode).value))
            locCode = UCase$(Replace(Replace(Replace(locCode, vbTab, ""), vbCr, ""), vbLf, ""))
            If Len(locCode) = 0 Then GoTo NEXT_LOC

            On Error GoTo EH_LOC

            stepName = "PATHS: " & locCode
            folderOut = basePath & locCode & "\REPORTE DE INCIDENCIAS DE NOMINA\"
            fullPath = folderOut & "Incidencias_" & locCode & ".xlsm"
            EnsureFolderExists folderOut

            stepName = "COPY TEMPLATE: " & locCode
            tmpPath = Environ$("TEMP") & "\__TMP__INC_" & locCode & "_" & Format(Now, "hhnnss") & ".xlsm"
            On Error Resume Next
            Kill tmpPath
            On Error GoTo EH_LOC

            ThisWorkbook.SaveCopyAs tmpPath
            Set wbNew = Workbooks.Open(fileName:=tmpPath, UpdateLinks:=False)

            '1) INYECTAR EMPLEADOS (desde el MISMO wbNew)
            stepName = "INJECT EMP " & locCode
            InjectEmployees_FromTemplateTables wbNew, locCode

            '2) CONFIG HIJO (evita rutas https / dependencia)
            stepName = "CONFIG " & locCode
            Dim locName As String, locDisplay As String
            
            'Si tu tabla trae estos campos, los leemos; si no, usamos el código como fallback
            locName = Nz(GetLocField(loLoc, r, "LocationName"), locCode)
            locDisplay = Nz(GetLocField(loLoc, r, "LocationDisplay"), locName)
            
            SetConfig_InWorkbook wbNew, "LocationCode", locCode
            SetConfig_InWorkbook wbNew, "LocationName", locName
            SetConfig_InWorkbook wbNew, "LocationDisplay", locDisplay
            SetConfig_InWorkbook wbNew, "IsTemplate", "0"


            'limpiar rutas que vienen como https en archivos abiertos desde SharePoint/Web
            SetConfig_InWorkbook wbNew, "MasterDBPath", ""
            SetConfig_InWorkbook wbNew, "EmployeeDBPath", ""
            SetConfig_InWorkbook wbNew, "GerentesRootPath", basePath

            stepName = "SAVE TMP " & locCode
            wbNew.Save
            wbNew.Close SaveChanges:=False
            Set wbNew = Nothing
            
            stepName = "COPY FINAL " & locCode
            SafeOverwriteCopyFile tmpPath, fullPath


            okCount = okCount + 1
            LogMem logText, "OK | " & locCode

        End If

NEXT_LOC:
        DoEvents
        On Error GoTo EH
    Next r

SAFEEXIT:
    On Error Resume Next
    If Not wbNew Is Nothing Then wbNew.Close SaveChanges:=False

    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True

    MsgBox "Generación terminada." & vbCrLf & _
           "OK=" & okCount & " | ERR=" & errCount & vbCrLf & vbCrLf & _
           "LOG (primeras líneas):" & vbCrLf & Left$(logText, 1800), vbInformation
    Exit Sub

EH_LOC:
    errCount = errCount + 1
    LogMem logText, "ERR | " & locCode & " | Step: " & stepName & " | " & err.Number & " - " & err.Description
    On Error Resume Next
    If Not wbNew Is Nothing Then wbNew.Close SaveChanges:=False
    Set wbNew = Nothing
    Resume NEXT_LOC

Private Sub EnsureFolderExists(ByVal folderPath As String)
    On Error Resume Next
    Dim fso As Object: Set fso = CreateObject("Scripting.FileSystemObject")

    folderPath = Replace(folderPath, "/", "\")
    If Right$(folderPath, 1) = "\" Then folderPath = Left$(folderPath, Len(folderPath) - 1)

    Dim parts() As String, i As Long, cur As String
    parts = Split(folderPath, "\")
    If UBound(parts) < 0 Then Exit Sub

    cur = parts(0)
    For i = 1 To UBound(parts)
        If parts(i) <> "" Then
            cur = cur & "\" & parts(i)
            If Not fso.FolderExists(cur) Then fso.CreateFolder cur
        End If
    Next i
End Sub

'Filtra por valor en columna idxCol (rápido, sin AutoFilter)
Private Function GetFilteredTableData_Fast(ByVal lo As ListObject, ByVal idxCol As Long, ByVal matchValue As String) As Variant
    On Error GoTo EH

    If lo.DataBodyRange Is Nothing Then
        GetFilteredTableData_Fast = Empty
        Exit Function
    End If

    Dim src As Variant
    src = lo.DataBodyRange.value

    Dim i As Long, j As Long, cnt As Long, outRow As Long, nCols As Long
    nCols = UBound(src, 2)

    cnt = 0
    For i = 1 To UBound(src, 1)
        If Trim$(CStr(src(i, idxCol))) = matchValue Then cnt = cnt + 1
    Next i

    If cnt = 0 Then
        GetFilteredTableData_Fast = Empty
        Exit Function
    End If

    Dim outArr() As Variant
    ReDim outArr(1 To cnt, 1 To nCols)

    outRow = 0
    For i = 1 To UBound(src, 1)
        If Trim$(CStr(src(i, idxCol))) = matchValue Then
            outRow = outRow + 1
            For j = 1 To nCols
                outArr(outRow, j) = src(i, j)
            Next j
        End If
    Next i

    GetFilteredTableData_Fast = outArr
    Exit Function

EH:
    GetFilteredTableData_Fast = Empty
End Function

Public Function CleanPath(ByVal p As String) As String
    p = Trim$(p)
    p = Replace(p, Chr$(34), "") 'quita comillas "

    'Si es URL, NO la toques (si quieres, puedes regresarla tal cual o vaciarla)
    If LCase$(Left$(p, 8)) = "https://" Or LCase$(Left$(p, 7)) = "http://" Then
        CleanPath = p
        Exit Function
    End If

    'normaliza rutas locales
    p = Replace(p, "/", "\")
    Do While InStr(p, "\\") > 0
        p = Replace(p, "\\", "\")
    Loop
    CleanPath = p
End Function

Private Sub SafeOverwriteCopyFile(ByVal src As String, ByVal dst As String)
    On Error GoTo EH

    src = CleanPath(src)
    dst = CleanPath(dst)

    If Len(Dir(src)) = 0 Then err.Raise vbObjectError + 701, , "No existe src: " & src

    EnsureFolderExists Left$(dst, InStrRev(dst, "\") - 1)

    On Error Resume Next
    If Len(Dir(dst)) > 0 Then Kill dst
    On Error GoTo EH

    FileCopy src, dst
    Exit Sub

EH:
    err.Raise vbObjectError + 799, , "SafeOverwriteCopyFile: " & err.Number & " - " & err.Description & vbCrLf & _
                                    "SRC: " & src & vbCrLf & "DST: " & dst
End Sub

    ' MODO EXTERNO (solo si es ruta local real)
    '========================================================
    If Len(dbTable) = 0 Then
        MsgBox "Falta configurar EmployeeDBTable en Config.", vbCritical

    End If

    Application.DisplayAlerts = False
    Application.EnableEvents = False

Private Function IsWebPath(ByVal p As String) As Boolean
    Dim s As String
    s = LCase$(Trim$(p))

    'Normaliza para atrapar "https:\..." y "https://..."
    s = Replace(s, "/", "\")
    
    If Left$(s, 4) = "http" Then IsWebPath = True: Exit Function
    If Left$(s, 5) = "https" Then IsWebPath = True: Exit Function
    If InStr(s, "sharepoint.com") > 0 Then IsWebPath = True: Exit Function
    If InStr(s, "onedrive") > 0 Then IsWebPath = True: Exit Function
    If InStr(s, "://") > 0 Then IsWebPath = True: Exit Function

    IsWebPath = False
End Function

' UID OFICIAL (INCIDENCIAS POR DÍA)
' Formato: LOC|NUMEMP|AÑO|MM|TIPO|PERIODO|DIA
'==========================
Public Function BuildUID_Incidencia( _
    ByVal loc As String, ByVal numEmp As Long, ByVal anio As Long, _
    ByVal mes As Long, ByVal tipoPeriodo As String, ByVal periodo As Long, _
    ByVal dia As Long) As String

    BuildUID_Incidencia = UCase$(Trim$(loc)) & "|" & _
                          CStr(numEmp) & "|" & _
                          CStr(anio) & "|" & Format$(mes, "00") & "|" & _
                          UCase$(Trim$(tipoPeriodo)) & "|" & _
                          CStr(periodo) & "|" & CStr(dia)
End Function

'====================================================
' modMantenimientoMatrices
' Limpia hojas de matriz M_... por locación,
' conservando SOLO:
'   - periodo actual (según fecha hoy)
'   - periodo anterior inmediato
' Para SEMANAL y QUINCENAL.
'====================================================

Public Sub LimpiarMatricesRelevantes(Optional ByVal locCode As String = "", Optional ByVal pwdLibro As String = "AVASA")

    Dim loc As String
    loc = Trim$(locCode)
    If loc = "" Then loc = Trim$(gLoc)
    If loc = "" Then Exit Sub

    '1) Calcular "targets" (actual y anterior) para semanal y quincenal
    Dim y As Long, m As Long
    Dim tipo As String, per As Long

    Dim keepDict As Object
    Set keepDict = CreateObject("Scripting.Dictionary") 'key: sheetname => True

    '--- QUINCENAL ---
    y = Year(Date): m = Month(Date)
    tipo = "QUINCENAL"
    per = PeriodoActualPorFecha(tipo, Date)

    AddKeep keepDict, BuildNombreMatriz(loc, y, m, tipo, per)
    AddKeep keepDict, NombreMatrizAnterior(loc, y, m, tipo, per)

    '--- SEMANAL ---
    y = Year(Date): m = Month(Date)
    tipo = "SEMANAL"
    per = PeriodoActualPorFecha(tipo, Date)

    AddKeep keepDict, BuildNombreMatriz(loc, y, m, tipo, per)
    AddKeep keepDict, NombreMatrizAnterior(loc, y, m, tipo, per)

    '2) Borrar cualquier M_ de esta loc que NO esté en keep
    Dim ws As Worksheet
    Dim ok As Boolean
    Dim pLoc As String, pAnio As Long, pMes As Long, pTipo As String, pNumPer As Long

    Dim estabaProtegido As Boolean
    estabaProtegido = ThisWorkbook.ProtectStructure

    If estabaProtegido Then
        On Error Resume Next
        ThisWorkbook.Unprotect pwdLibro
        On Error GoTo 0
    End If

    Application.DisplayAlerts = False

    Dim toDelete As Collection
    Set toDelete = New Collection

    For Each ws In ThisWorkbook.Worksheets
        If Left$(ws.Name, 2) = "M_" Then
            ok = TryParseNombreMatriz(ws.Name, pLoc, pAnio, pMes, pTipo, pNumPer)
            If ok Then
                If UCase$(pLoc) = UCase$(loc) Then
                    If Not keepDict.Exists(ws.Name) Then
                        toDelete.Add ws.Name
                    End If
                End If
            End If
        End If
    Next ws

    Dim nm As Variant
    For Each nm In toDelete
        On Error Resume Next
        ThisWorkbook.Worksheets(CStr(nm)).Delete
        On Error GoTo 0
    Next nm

    Application.DisplayAlerts = True

    If estabaProtegido Then
        On Error Resume Next
        ThisWorkbook.Protect Password:=pwdLibro, Structure:=True
        On Error GoTo 0
    End If

End Sub

'========================
' Helpers (KEEP)
'========================
Private Sub AddKeep(ByVal dict As Object, ByVal sheetName As String)
    If Len(sheetName) = 0 Then Exit Sub
    If Not dict.Exists(sheetName) Then dict.Add sheetName, True
End Sub

Private Function BuildNombreMatriz(ByVal loc As String, ByVal anio As Long, ByVal mes As Long, ByVal tipo As String, ByVal numPeriodo As Long) As String
    Dim suf As String
    If UCase$(tipo) = "SEMANAL" Then
        suf = "S" & CStr(numPeriodo)
    Else
        suf = "Q" & CStr(numPeriodo)
    End If

    BuildNombreMatriz = "M_" & loc & "_" & CStr(anio) & "_" & Format$(mes, "00") & "_" & suf

    'Si tu NombreHojaMatriz truncó a 31, aquí igualamos para coincidir
    If Len(BuildNombreMatriz) > 31 Then BuildNombreMatriz = Left$(BuildNombreMatriz, 31)
End Function

Private Function NombreMatrizAnterior(ByVal loc As String, ByVal anio As Long, ByVal mes As Long, ByVal tipo As String, ByVal numPeriodo As Long) As String

    Dim y2 As Long, m2 As Long, p2 As Long
    y2 = anio: m2 = mes: p2 = numPeriodo - 1

    If UCase$(tipo) = "QUINCENAL" Then
        If p2 < 1 Then
            'ir al mes anterior y quedarse con Q2
            BackOneMonth y2, m2
            p2 = 2
        End If

    ElseIf UCase$(tipo) = "SEMANAL" Then
        If p2 < 1 Then
            'ir al mes anterior y quedarse con S4
            BackOneMonth y2, m2
            p2 = 4
        End If
    End If

    NombreMatrizAnterior = BuildNombreMatriz(loc, y2, m2, tipo, p2)
End Function

Private Sub BackOneMonth(ByRef anio As Long, ByRef mes As Long)
    mes = mes - 1
    If mes < 1 Then
        mes = 12
        anio = anio - 1
    End If
End Sub

'========================
' Periodo actual por fecha
'========================
Private Function PeriodoActualPorFecha(ByVal tipo As String, ByVal dt As Date) As Long
    Dim d As Long, uDia As Long
    d = Day(dt)
    uDia = Day(DateSerial(Year(dt), Month(dt) + 1, 0))

    Select Case UCase$(tipo)
        Case "QUINCENAL"
            If d <= 15 Then
                PeriodoActualPorFecha = 1
            Else
                PeriodoActualPorFecha = 2
            End If

        Case "SEMANAL"
            If d <= 7 Then
                PeriodoActualPorFecha = 1
            ElseIf d <= 14 Then
                PeriodoActualPorFecha = 2
            ElseIf d <= 21 Then
                PeriodoActualPorFecha = 3
            Else
                PeriodoActualPorFecha = 4
            End If

        Case Else
            PeriodoActualPorFecha = 0
    End Select
End Function

'========================
' Parseo robusto del nombre
' Soporta loc con underscores.
' Formato esperado:
'   M_<LOC>_<YYYY>_<MM>_<Q|S><N>
' Ej: M_CUU_2025_12_Q1
'========================
Public Function TryParseNombreMatriz( _
        ByVal sheetName As String, _
        ByRef loc As String, _
        ByRef anio As Long, _
        ByRef mes As Long, _
        ByRef tipo As String, _
        ByRef numPeriodo As Long) As Boolean

    On Error GoTo Fail

    Dim re As Object, m As Object
    Set re = CreateObject("VBScript.RegExp")

    re.Pattern = "^M_(.+)_(\d{4})_(\d{2})_([QS])(\d+)$"
    re.IgnoreCase = True
    re.Global = False

    If Not re.Test(sheetName) Then GoTo Fail

    Set m = re.Execute(sheetName)(0)

    loc = CStr(m.SubMatches(0))
    anio = CLng(m.SubMatches(1))
    mes = CLng(m.SubMatches(2))

    Dim t As String
    t = UCase$(CStr(m.SubMatches(3)))
    If t = "Q" Then
        tipo = "QUINCENAL"
    Else
        tipo = "SEMANAL"
    End If

    numPeriodo = CLng(m.SubMatches(4))

    TryParseNombreMatriz = True
    Exit Function

Fail:
    TryParseNombreMatriz = False
End Function

Private dictFestivos As Object          ' key: CLng(Date)  value: TRUE

Public Sub AplicarCalendarioAMatriz(ByVal wsMatriz As Worksheet)

    Set rngFechas = GetHeaderFechasRange(wsMatriz)
If dictFestivos Is Nothing Then CargarFestivosEnMemoria


    If Not rngFechas Is Nothing Then
        PintarDiasEspeciales wsMatriz, rngFechas
    End If

'==============================
'  CORE
'==============================
Public Function GetDiaTipo(ByVal dt As Date) As String
    If EsFestivo(dt) Then
        GetDiaTipo = "DF"
    ElseIf Weekday(dt, vbMonday) = 7 Then
        GetDiaTipo = "PD"
    Else
        GetDiaTipo = "NORMAL"
    End If
End Function

Public Function EsFestivo(ByVal dt As Date) As Boolean
    On Error Resume Next

    If dictFestivos Is Nothing Then
        CargarFestivosEnMemoria
    End If

    EsFestivo = False
    If Not dictFestivos Is Nothing Then
        EsFestivo = dictFestivos.Exists(CLng(DateValue(dt)))
    End If
End Function

Public Function GetFestivoNombre(ByVal dt As Date) As String
    GetFestivoNombre = ""
    If dictFestivoNombre Is Nothing Then Exit Function

    Dim k As Long
    k = CLng(DateValue(dt))

    If dictFestivoNombre.Exists(k) Then
        GetFestivoNombre = CStr(dictFestivoNombre(k))
    End If
End Function

'==============================
'  MATRIZ
'==============================
Private Function GetHeaderFechasRange(ByVal ws As Worksheet) As Range
    Dim startCol As Long, c As Long, lastCol As Long

    startCol = 9 ' Columna I
    lastCol = startCol

    ' Avanzar mientras haya números (días)
    For c = startCol To ws.Columns.count
        If IsNumeric(ws.Cells(2, c).value) Then
            lastCol = c
        Else
            Exit For
        End If
    Next c

    If lastCol < startCol Then Exit Function

    Set GetHeaderFechasRange = ws.Range(ws.Cells(2, startCol), ws.Cells(2, lastCol))
End Function

Private Sub PintarDiasEspeciales(ByVal ws As Worksheet, ByVal headerRange As Range)
    Dim c As Range
    Dim tipo As String
    Dim fechaReal As Date
    Dim dia As Long

    'IMPORTANTE: si hay Formato Condicional, puede estar sobre-escribiendo el color que pongas por VBA
    On Error Resume Next
    headerRange.FormatConditions.Delete
    On Error GoTo 0

    For Each c In headerRange.Cells

        If IsNumeric(c.value) Then
            dia = CLng(c.value)

            'Color base SIEMPRE (amarillo)
            c.Interior.Color = RGB(240, 200, 80)
            c.Font.Bold = True
            c.Font.Italic = False

            'Fecha real del periodo
            fechaReal = DateSerial(gAnio, gMes, dia)
            tipo = GetDiaTipo(fechaReal)

            Select Case tipo
                Case "DF"
                    'Festivo
                    c.Interior.Color = RGB(255, 180, 180) 'rojo suave

                Case "PD"
                    'Domingo
                    c.Interior.Color = RGB(220, 220, 220) 'gris suave
            End Select
        End If

    Next c
End Sub

'==============================
'  CARGA FESTIVOS A MEMORIA
'==============================
Public Sub CargarFestivosEnMemoria()
    Dim lo As ListObject
    Dim lr As ListRow
    Dim vFecha As Variant, vNombre As Variant, vActivo As Variant
    Dim f As Date, key As Long

    Set dictFestivos = CreateObject("Scripting.Dictionary")
    Set dictFestivoNombre = CreateObject("Scripting.Dictionary")

    Set lo = ThisWorkbook.Worksheets("Config").ListObjects("tblFestivos")

    For Each lr In lo.ListRows
        vFecha = lr.Range(1, 1).value  ' Fecha
        vNombre = lr.Range(1, 2).value ' Nombre
        vActivo = lr.Range(1, 5).value ' Activo

        If IsDate(vFecha) Then
            f = DateValue(CDate(vFecha))
            key = CLng(f)

            If EsVerdadero(vActivo) Then
                dictFestivos(key) = True
                dictFestivoNombre(key) = CStr(vNombre)
            End If
        End If
    Next lr
End Sub

Private Function EsVerdadero(ByVal v As Variant) As Boolean
    On Error GoTo Falso
    If IsError(v) Or IsEmpty(v) Then GoTo Falso

    If VarType(v) = vbBoolean Then
        EsVerdadero = (v = True)
        Exit Function
    End If

    If IsNumeric(v) Then
        EsVerdadero = (CLng(v) <> 0)
        Exit Function
    End If

    Select Case UCase$(Trim$(CStr(v)))
        Case "TRUE", "VERDADERO", "SI", "SÍ", "1", "X"
            EsVerdadero = True
        Case Else
            EsVerdadero = False
    End Select
    Exit Function

Falso:
    EsVerdadero = False
End Function

'=========================================
' Normalización
'=========================================
' Normaliza para comparar: mayúsculas, sin espacios, sin "/"
Public Function NormalizaCodigo(ByVal s As String) As String
    s = UCase$(Trim$(s))
    s = Replace$(s, " ", "")
    s = Replace$(s, "/", "")
    NormalizaCodigo = s
End Function

' Canoniza alias históricos y limpia basura
' - "T/D" -> "TD"
' - "FI"  -> "F"
' - "0"   -> "" (lo eliminamos)
Public Function CanonizarCodigo(ByVal codigo As String) As String
    Dim norm As String
    norm = NormalizaCodigo(codigo)

    Select Case norm
        Case "", "0"
            CanonizarCodigo = ""
        Case "TD", "T/D"
            CanonizarCodigo = "TD"
        Case "FI"
            CanonizarCodigo = "F"
        Case Else
            CanonizarCodigo = norm
    End Select
End Function

'=========================================
' Catálogo
'=========================================
' Recalcula la columna [Normalizado] = NormalizaCodigo([Codigo])
Public Sub Catalogo_RecalcularNormalizados()
    Dim lo As ListObject, lr As ListRow
    Dim idxCod As Long, idxNorm As Long

    Set lo = ThisWorkbook.Worksheets(WS_CFG).ListObjects(LO_CAT)
    If lo.DataBodyRange Is Nothing Then Exit Sub

    idxCod = lo.ListColumns("Codigo").Index
    idxNorm = lo.ListColumns("Normalizado").Index

    For Each lr In lo.ListRows
        lr.Range(1, idxNorm).value = NormalizaCodigo(CStr(lr.Range(1, idxCod).value))
    Next lr
End Sub

' Devuelve array 1D con códigos ACTIVOS (para dropdown)
Public Function GetCodigosActivos() As Variant
    Dim lo As ListObject
    Dim arr, out(), i As Long, n As Long
    Dim idxCod As Long, idxActivo As Long

    Set lo = ThisWorkbook.Worksheets(WS_CFG).ListObjects(LO_CAT)
    If lo.DataBodyRange Is Nothing Then Exit Function

    arr = lo.DataBodyRange.value
    idxCod = lo.ListColumns("Codigo").Index
    idxActivo = lo.ListColumns("Activo").Index

    ReDim out(1 To UBound(arr, 1)) 'máximo
    n = 0

    For i = 1 To UBound(arr, 1)
        If VBA.CBool(arr(i, idxActivo)) = True Then
            n = n + 1
            out(n) = CStr(arr(i, idxCod))
        End If
    Next i

    If n = 0 Then Exit Function
    ReDim Preserve out(1 To n)

    GetCodigosActivos = out
End Function

' Valida si un código (o alias) existe en catálogo y está ACTIVO
' OJO: permite "" (vacío) como válido
Public Function EsCodigoValido(ByVal codigo As String) As Boolean
    Dim lo As ListObject
    Dim rngNorm As Range
    Dim norm As String, r As Variant
    Dim idxActivo As Long

    norm = CanonizarCodigo(codigo)
    If Len(norm) = 0 Then
        EsCodigoValido = True
        Exit Function
    End If

    Set lo = ThisWorkbook.Worksheets(WS_CFG).ListObjects(LO_CAT)
    Set rngNorm = lo.ListColumns("Normalizado").DataBodyRange
    idxActivo = lo.ListColumns("Activo").Index

    r = Application.Match(norm, rngNorm, 0)
    If IsError(r) Then
        EsCodigoValido = False
    Else
        EsCodigoValido = VBA.CBool(lo.DataBodyRange.Cells(CLng(r), idxActivo).value)
    End If
End Function

'=========================================
' BD (opcional): canoniza toda la columna O
'=========================================
Public Sub BD_CanonizarCodigos()
    Dim ws As Worksheet
    Dim lastRow As Long, i As Long
    Dim v As String, canon As String

    Set ws = ThisWorkbook.Worksheets("BDIncidencias_Local")
    lastRow = ws.Cells(ws.Rows.count, "O").End(xlUp).Row ' O = CodigoInc

    Application.ScreenUpdating = False

    On Error Resume Next

    On Error GoTo 0

    Application.EnableEvents = True
    Application.ScreenUpdating = True

    MsgBox "Listo. Canonizados códigos en BD: columna O.", vbInformation

Public Function DescripcionDeCodigo(ByVal cod As String) As String
    cod = UCase$(Trim$(cod))
    If Len(cod) = 0 Then
        DescripcionDeCodigo = ""
        Exit Function
    End If

    If gDictCodDesc Is Nothing Then Set gDictCodDesc = BuildDictCodDesc()

    If Not gDictCodDesc Is Nothing Then
        If gDictCodDesc.Exists(cod) Then
            DescripcionDeCodigo = CStr(gDictCodDesc(cod))
            Exit Function
        End If
    End If

    'fallback: si no encuentra descripción, muestra el código
    DescripcionDeCodigo = cod
End Function

Private Function BuildDictCodDesc() As Object
    Dim dict As Object: Set dict = CreateObject("Scripting.Dictionary")
    Dim ws As Worksheet, lo As ListObject
    Dim cCod As Long, cDesc As Long
    Dim r As ListRow, k As String, v As String

    'Busca una tabla que tenga columnas tipo "Codigo" y "Descripcion"
    For Each ws In ThisWorkbook.Worksheets
        For Each lo In ws.ListObjects
            cCod = GetColIndex(lo, Array("CODIGO", "CODE", "INCIDENCIA", "CLAVE"))
            cDesc = GetColIndex(lo, Array("DESCRIPCION", "DESCRIPCIÓN", "NOMBRE", "DESCRIPCIONINCIDENCIA"))
            If cCod > 0 And cDesc > 0 Then
                For Each r In lo.ListRows
                    k = UCase$(Trim$(CStr(r.Range.Cells(1, cCod).value)))
                    v = Trim$(CStr(r.Range.Cells(1, cDesc).value))
                    If Len(k) > 0 And Len(v) > 0 Then dict(k) = v
                Next r
                Set BuildDictCodDesc = dict
                Exit Function
            End If
        Next lo
    Next ws

    'si no encuentra tabla, regresa dict vacío (no truena)
    Set BuildDictCodDesc = dict
End Function

Private Function GetColIndex(ByVal lo As ListObject, ByVal candidates As Variant) As Long
    Dim i As Long, nameU As String, c As Variant
    For i = 1 To lo.ListColumns.count
        nameU = UCase$(Trim$(lo.ListColumns(i).Name))
        For Each c In candidates
            If nameU = UCase$(CStr(c)) Then
                GetColIndex = i
                Exit Function
            End If
        Next c
    Next i
    GetColIndex = 0
End Function

'----------------------------------------------------
' Util: asegura que la ruta termine con "\"
'----------------------------------------------------
Private Function NormalizePath(ByVal p As String) As String
    p = Trim$(p)
    If Len(p) = 0 Then
        NormalizePath = ""
    ElseIf Right$(p, 1) = "\" Then
        NormalizePath = p
    Else
        NormalizePath = p & "\"
    End If
End Function

End Function

'----------------------------------------------------
' Exporta TODO el proyecto VBA a un solo TXT con timestamp
'----------------------------------------------------
Public Sub ExportarVBA_A_UnSoloTXT()

    Dim exportBase As String
    Dim exportFile As String
    Dim stamp As String
    Dim vbComp As Object
    Dim ff As Integer
    Dim i As Long

    exportBase = NormalizePath(GetExportPath())
    If exportBase = "" Then
        MsgBox "GetExportPath() está vacío.", vbCritical
        Exit Sub
    End If

    If Dir(exportBase, vbDirectory) = "" Then
        MsgBox "La carpeta no existe:" & vbCrLf & exportBase, vbCritical
        Exit Sub
    End If

    stamp = GetStamp()
    exportFile = exportBase & "VBA_COMPLETO_" & stamp & ".txt"

    ff = FreeFile
    Open exportFile For Output As #ff

    ' Encabezado
    Print #ff, "===================================================="
    Print #ff, "VBA_COMPLETO — Export automático"
    Print #ff, "Fecha: " & Format(Now, "yyyy-mm-dd hh:nn:ss")
    Print #ff, "Workbook: " & ThisWorkbook.Name
    Print #ff, "===================================================="
    Print #ff, vbCrLf

    ' Exportación por componente (pegado de código)
    For Each vbComp In ThisWorkbook.VBProject.VBComponents

        'Opcional: saltar componentes con "_" al inicio
        If Left$(vbComp.Name, 1) = "_" Then GoTo NextCompTXT

        Print #ff, "========================================"
        Print #ff, "COMPONENTE: " & vbComp.Name
        Print #ff, "TIPO: " & vbComp.Type
        Print #ff, "========================================"

        If vbComp.CodeModule.CountOfLines > 0 Then
            For i = 1 To vbComp.CodeModule.CountOfLines
                Print #ff, vbComp.CodeModule.Lines(i, 1)
            Next i
        End If

        Print #ff, vbCrLf

NextCompTXT:
    Next vbComp

    Close #ff

    MsgBox "Archivo creado correctamente:" & vbCrLf & exportFile, vbInformation

End Sub

    exportBase = NormalizePath(GetExportPath())
    If exportBase = "" Then
        MsgBox "GetExportPath() está vacío.", vbCritical
        Exit Sub
    End If

    If Dir(exportBase, vbDirectory) = "" Then
        MsgBox "La carpeta no existe:" & vbCrLf & exportBase, vbCritical
        Exit Sub
    End If

    stamp = GetStamp()
    exportPath = exportBase & "VBA_EXPORT_" & stamp & "\"
    EnsureFolder exportPath

    'INDEX.txt
    ff = FreeFile
    Open exportPath & "INDEX.txt" For Output As #ff
    Print #ff, "Export VBA - " & Format(Now, "yyyy-mm-dd hh:nn:ss")
    Print #ff, "Workbook: " & ThisWorkbook.Name
    Print #ff, "Folder: " & exportPath
    Print #ff, String(70, "-")
    Print #ff, "Name | Type | File | Result"
    Print #ff, String(70, "-")

    exportOK = True

    For Each vbComp In ThisWorkbook.VBProject.VBComponents

        'Opcional: saltar componentes con "_" al inicio
        If Left$(vbComp.Name, 1) = "_" Then GoTo NextComp

        ext = ExtByType(vbComp.Type)
        outFile = exportPath & vbComp.Name & ext

        err.Clear
        On Error Resume Next
        vbComp.Export outFile
        If err.Number <> 0 Then
            exportOK = False
            errMsg = "ERR " & err.Number & ": " & err.Description
            Print #ff, vbComp.Name & " | " & vbComp.Type & " | " & (vbComp.Name & ext) & " | " & errMsg
        Else
            Print #ff, vbComp.Name & " | " & vbComp.Type & " | " & (vbComp.Name & ext) & " | OK"
        End If
        On Error GoTo 0

NextComp:
    Next vbComp

    Close #ff

    If exportOK Then
        MsgBox "Exportación por componentes completada en:" & vbCrLf & exportPath, vbInformation
    Else
        MsgBox "Exportación terminó con algunos errores." & vbCrLf & _
               "Revisa INDEX.txt en:" & vbCrLf & exportPath, vbExclamation
    End If

End Sub

Private Function NzStr(ByVal v As Variant) As String
    If IsError(v) Or IsNull(v) Or Len(Trim$(CStr(v))) = 0 Then NzStr = "" Else NzStr = Trim$(CStr(v))
End Function

Private Function NzLng(ByVal v As Variant, Optional ByVal def As Long = 0) As Long
    On Error GoTo Fail

    If IsError(v) Or IsNull(v) Or IsEmpty(v) Then
        NzLng = def
        Exit Function
    End If

    Dim s As String
    s = Trim$(CStr(v))

    If Len(s) = 0 Then
        NzLng = def
    ElseIf IsNumeric(s) Then
        NzLng = CLng(CDbl(s)) 'soporta "12.0" también
    ElseIf IsDate(v) Then
        NzLng = CLng(CDate(v))
    Else
        NzLng = def
    End If

    Exit Function
Fail:
    NzLng = def
End Function

Private Function FindRow(ByVal loc As String, ByVal anio As Long, ByVal mes As Long, _
                         ByVal tipo As String, ByVal periodo As Long) As ListRow
    Dim lo As ListObject, lr As ListRow
    Set lo = GetLO()

    loc = UCase$(Trim$(loc))
    tipo = UCase$(Trim$(tipo))

    For Each lr In lo.ListRows
        If UCase$(NzStr(lr.Range(1, lo.ListColumns("LocCode").Index).value)) = loc _
           And NzLng(lr.Range(1, lo.ListColumns("Anio").Index).value) = anio _
           And NzLng(lr.Range(1, lo.ListColumns("Mes").Index).value) = mes _
           And UCase$(NzStr(lr.Range(1, lo.ListColumns("TipoPeriodo").Index).value)) = tipo _
           And NzLng(lr.Range(1, lo.ListColumns("Periodo").Index).value) = periodo Then
            Set FindRow = lr
            Exit Function
        End If
    Next lr
End Function

' Public API
'========================
Public Sub EnsurePeriodoRow(ByVal loc As String, ByVal anio As Long, ByVal mes As Long, _
                            ByVal tipo As String, ByVal periodo As Long)
    On Error GoTo Fail

    Dim lo As ListObject, lr As ListRow
    Set lo = GetLO()

    Set lr = FindRow(loc, anio, mes, tipo, periodo)
    If Not lr Is Nothing Then Exit Sub

    Set lr = lo.ListRows.Add
    lr.Range(1, lo.ListColumns("LocCode").Index).value = UCase$(Trim$(loc))
    lr.Range(1, lo.ListColumns("Anio").Index).value = anio
    lr.Range(1, lo.ListColumns("Mes").Index).value = mes
    lr.Range(1, lo.ListColumns("TipoPeriodo").Index).value = UCase$(Trim$(tipo))
    lr.Range(1, lo.ListColumns("Periodo").Index).value = periodo
    lr.Range(1, lo.ListColumns("Status").Index).value = "CAPTURA"

    On Error Resume Next
    lr.Range(1, lo.ListColumns("UpdatedAt").Index).value = Now
    On Error GoTo 0

Fail:
    'Si falla, no rompemos nada: el sistema seguirá con la lógica actual por fecha
End Sub

Public Function TerminalToCCCode(ByVal terminal As String) As String
    terminal = UCase$(Trim$(terminal))
    'Ej: "can50", "CAN50-01" -> "CAN50"
    If Len(terminal) >= 5 Then
        TerminalToCCCode = Left$(terminal, 5)
    Else
        TerminalToCCCode = terminal
    End If
End Function

'--------------------------------------------
' ¿Esta locación maneja precarga por checador?
' Lee columna: TieneChecador en tblLocaciones (hoja Locaciones)
'--------------------------------------------
Public Function LocacionTieneChecador(ByVal locCode As String) As Boolean
    On Error GoTo Fail

    Dim ws As Worksheet, lo As ListObject
    Dim colLoc As Long, colFlag As Long
    Dim r As Long, lastR As Long
    Dim vLoc As String

    locCode = UCase$(Trim$(locCode))
    If Len(locCode) = 0 Then Exit Function

    'HOJA CORRECTA
    Set ws = ThisWorkbook.Worksheets("Locaciones")
    Set lo = ws.ListObjects("tblLocaciones")

    'COLUMNA CÓDIGO (en tu captura es LocationCode)
    If ListColumnExists(lo, "LocationCode") Then
        colLoc = lo.ListColumns("LocationCode").Index
    ElseIf ListColumnExists(lo, "LocCode") Then
        colLoc = lo.ListColumns("LocCode").Index
    Else
        GoTo Fail
    End If

    'BANDERA
    If Not ListColumnExists(lo, "TieneChecador") Then GoTo Fail
    colFlag = lo.ListColumns("TieneChecador").Index

    lastR = lo.DataBodyRange.Rows.count

    For r = 1 To lastR
        vLoc = UCase$(Trim$(CStr(lo.DataBodyRange.Cells(r, colLoc).value)))
        If vLoc = locCode Then
            LocacionTieneChecador = ValToBool(lo.DataBodyRange.Cells(r, colFlag).value)
            Exit Function
        End If
    Next r

    Exit Function

Fail:
    LocacionTieneChecador = False
End Function

Private Function ListColumnExists(ByVal lo As ListObject, ByVal colName As String) As Boolean
    On Error GoTo Nope
    Dim tmp As Long
    tmp = lo.ListColumns(colName).Index
    ListColumnExists = True
    Exit Function
Nope:
    ListColumnExists = False
End Function

Public Function ValToBool(ByVal v As Variant) As Boolean
    On Error GoTo EH

    'Boolean directo
    If VarType(v) = vbBoolean Then
        ValToBool = CBool(v)
        Exit Function
    End If

    Dim s As String
    s = UCase$(Trim$(CStr(v)))

    If Len(s) = 0 Then
        ValToBool = False
        Exit Function
    End If

    'Acepta varias formas típicas
    Select Case s
        Case "TRUE", "VERDADERO", "1", "SI", "S", "YES", "Y", "X"
            ValToBool = True
        Case "FALSE", "FALSO", "0", "NO", "N"
            ValToBool = False
        Case Else
            'si es número cualquiera
            If IsNumeric(s) Then ValToBool = (CDbl(s) <> 0) Else ValToBool = False
    End Select

    Exit Function
EH:
    ValToBool = False
End Function

LogStart "PrecargarChecador"


    On Error GoTo Fail

    '-----------------------------------------
    ' 0) Validar periodo abierto
    '-----------------------------------------
    Dim msg As String
    If Not modSeguridadIncidencias.PermiteEdicionPeriodo(msg) Then
        MsgBox "No se puede precargar porque el " & vbCrLf & msg, vbExclamation
        Exit Sub
    End If

    '-----------------------------------------

    '-----------------------------------------

    Application.DisplayAlerts = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual
    Application.StatusBar = "Leyendo checador..."
    DoEvents

    ' 4) LEER CHECADOR -> dict (UserID|yyyymmdd => "1|1")
    '========================================
    Dim dict As Object: Set dict = CreateObject("Scripting.Dictionary")

    Dim wb As Workbook, ws As Worksheet
    Set wb = Workbooks.Open(CStr(filePath), ReadOnly:=True)
    Set ws = wb.Worksheets(1)

    Dim lastRow As Long, r As Long
    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row

    Dim userId As String, estado As String, terminal As String
    Dim fechaDT As Date, fecha As Date
    Dim cc As String, locFound As String
    Dim key As String, ymd As String, v As String
    Dim hasIn As String, hasOut As String
    Dim tmp() As String

    For r = 2 To lastRow
        userId = Trim$(CStr(ws.Cells(r, "A").value))
        If Len(userId) = 0 Then GoTo NextR

        If Not IsDate(ws.Cells(r, "C").value) Then GoTo NextR
        fechaDT = CDate(ws.Cells(r, "C").value)
        fecha = DateValue(fechaDT)
        If fecha < fIni Or fecha > fFin Then GoTo NextR

        estado = UCase$(Trim$(CStr(ws.Cells(r, "E").value)))
        terminal = CStr(ws.Cells(r, "F").value)

        cc = modCatalogos.TerminalToCCCode(terminal)
        locFound = modCatalogos.CCCodeToLocCode(cc)
        If locFound = "" Then GoTo NextR
        If UCase$(locFound) <> UCase$(gLoc) Then GoTo NextR

        ymd = Format$(fecha, "yyyymmdd")
        key = userId & "|" & ymd

        If dict.Exists(key) Then
            v = dict(key)
        Else
            v = "0|0"
        End If

        tmp = Split(v, "|")
        hasIn = tmp(0)
        hasOut = tmp(1)

        If estado = "ENTRADA" Then hasIn = "1"
        If estado = "SALIDA" Then hasOut = "1"

        dict(key) = hasIn & "|" & hasOut

NextR:
    Next r

    wb.Close SaveChanges:=False

    '========================================
    ' 5) INDEXAR BDIncidencias_Local (solo periodo actual + gLoc)
    '    idx(kbd)=row
    '========================================
    Application.StatusBar = "Indexando BDIncidencias_Local..."
    DoEvents

    Dim wsBD As Worksheet: Set wsBD = ThisWorkbook.Worksheets("BDIncidencias_Local")

    'Asegurar editable
    On Error Resume Next
    wsBD.Unprotect "AVASA"
    wsBD.Unprotect "IncidenciasAVASA"
    On Error GoTo 0

    Dim lastBD As Long: lastBD = wsBD.Cells(wsBD.Rows.count, "A").End(xlUp).Row
    If lastBD < 2 Then lastBD = 1

    Dim idx As Object: Set idx = CreateObject("Scripting.Dictionary")
    Dim i As Long, dia As Long
    Dim kbd As String, loc As String, emp As String, tipo As String

    For i = 2 To lastBD
        loc = CStr(wsBD.Cells(i, "A").value)
        If UCase$(loc) <> UCase$(gLoc) Then GoTo NextBD

        If CLng(val(wsBD.Cells(i, "I").value)) <> gAnio Then GoTo NextBD

        If CLng(val(wsBD.Cells(i, "L").value)) <> gPeriodo Then GoTo NextBD

        emp = CStr(wsBD.Cells(i, "C").value)
        dia = CLng(val(wsBD.Cells(i, "M").value))

        kbd = UCase$(gLoc) & "|" & emp & "|" & gAnio & "|" & gMes & "|" & UCase$(gTipoPeriodo) & "|" & gPeriodo & "|" & dia

    DoEvents

    Dim newRows As Collection: Set newRows = New Collection
    Dim kk As Variant, parts As Variant
    Dim cod As String, rowBD As Long
    Dim captPor As String
    Dim uid As String
    Dim nextID As Long

    Dim countSkipManual As Long, countSkipBadEmp As Long

        parts = Split(CStr(kk), "|")

        ymd = parts(1)

        If Not IsNumeric(userId) Then
            countSkipBadEmp = countSkipBadEmp + 1
            GoTo NextK
        End If

        fecha = DateSerial(CInt(Left$(ymd, 4)), CInt(Mid$(ymd, 5, 2)), CInt(Right$(ymd, 2)))
        dia = Day(fecha)

        If idx.Exists(kbd) Then
            rowBD = CLng(idx(kbd))

            captPor = UCase$(Trim$(CStr(wsBD.Cells(rowBD, "R").value)))

            'Si ya hay algo manual, NO tocar
            If captPor <> "" And captPor <> "CHECADOR" Then
                countSkipManual = countSkipManual + 1
                GoTo NextK
            End If

            'Si estaba vacío, lo "relleno"
            If Len(Trim$(CStr(wsBD.Cells(rowBD, "O").value))) = 0 Then
                wsBD.Cells(rowBD, "O").value = cod
                wsBD.Cells(rowBD, "R").value = "CHECADOR"
                wsBD.Cells(rowBD, "S").value = Now
                wsBD.Cells(rowBD, "T").value = "CHECADOR"
                wsBD.Cells(rowBD, "N").value = fecha

                If Len(Trim$(CStr(wsBD.Cells(rowBD, "W").value))) = 0 Then
                    uid = modUID.BuildUID_Incidencia(gLoc, CLng(userId), gAnio, gMes, gTipoPeriodo, gPeriodo, dia)
                    wsBD.Cells(rowBD, "W").value = uid
                End If
                If Len(Trim$(CStr(wsBD.Cells(rowBD, "U").value))) = 0 Then
                    wsBD.Cells(rowBD, "U").value = nextID
                    nextID = nextID + 1
                End If

                countFillEmpty = countFillEmpty + 1
            Else
                'Ya era checador -> actualizar (si cambió)
                If UCase$(Trim$(CStr(wsBD.Cells(rowBD, "O").value))) <> UCase$(cod) Then
                    wsBD.Cells(rowBD, "O").value = cod
                End If
                wsBD.Cells(rowBD, "R").value = "CHECADOR"
                wsBD.Cells(rowBD, "S").value = Now
                wsBD.Cells(rowBD, "T").value = "CHECADOR"
                If Len(Trim$(CStr(wsBD.Cells(rowBD, "N").value))) = 0 Then wsBD.Cells(rowBD, "N").value = fecha
                If Len(Trim$(CStr(wsBD.Cells(rowBD, "W").value))) = 0 Then
                    uid = modUID.BuildUID_Incidencia(gLoc, CLng(userId), gAnio, gMes, gTipoPeriodo, gPeriodo, dia)
                    wsBD.Cells(rowBD, "W").value = uid
                End If
                If Len(Trim$(CStr(wsBD.Cells(rowBD, "U").value))) = 0 Then
                    wsBD.Cells(rowBD, "U").value = nextID
                    nextID = nextID + 1
                End If

                countUpdateChecador = countUpdateChecador + 1
            End If

        Else
            'Insert masivo mínimo útil
            Dim arr(1 To 12) As Variant
            arr(1) = gLoc                         'A
            arr(2) = CLng(userId)                 'C
            arr(3) = gAnio                        'I
            arr(4) = gMes                         'J
            arr(5) = UCase$(gTipoPeriodo)         'K  (normalizado)
            arr(6) = gPeriodo                     'L
            arr(7) = dia                          'M
            arr(8) = fecha                        'N
            arr(9) = cod                          'O
            arr(10) = "CHECADOR"                  'R
            arr(11) = Now                         'S
            arr(12) = "CHECADOR"                  'T
            newRows.Add arr

            countInsert = countInsert + 1
        End If

NextK:
    Next kk

    '========================================
    ' 7) INSERT MASIVO (mínimo útil)
    '========================================
    Dim n As Long: n = newRows.count
    If n > 0 Then
        Application.StatusBar = "Insertando filas nuevas..."
        DoEvents

        Dim startRow As Long: startRow = wsBD.Cells(wsBD.Rows.count, "A").End(xlUp).Row + 1

        Dim outA() As Variant, outC() As Variant, outI() As Variant, outJ() As Variant
        Dim outK() As Variant, outL() As Variant, outM() As Variant, outN() As Variant
        Dim outO() As Variant, outR() As Variant, outS() As Variant, outT() As Variant
        Dim outU() As Variant, outW() As Variant

        ReDim outA(1 To n, 1 To 1)
        ReDim outC(1 To n, 1 To 1)
        ReDim outI(1 To n, 1 To 1)
        ReDim outJ(1 To n, 1 To 1)
        ReDim outK(1 To n, 1 To 1)
        ReDim outL(1 To n, 1 To 1)
        ReDim outM(1 To n, 1 To 1)
        ReDim outN(1 To n, 1 To 1)
        ReDim outO(1 To n, 1 To 1)
        ReDim outR(1 To n, 1 To 1)
        ReDim outS(1 To n, 1 To 1)
        ReDim outT(1 To n, 1 To 1)
        ReDim outU(1 To n, 1 To 1)
        ReDim outW(1 To n, 1 To 1)

        Dim j As Long, a As Variant
        Dim empNum As Long

        For j = 1 To n
            a = newRows(j)

            empNum = CLng(a(2))
            dia = CLng(a(7))
            fecha = CDate(a(8))

            outA(j, 1) = a(1)
            outC(j, 1) = empNum
            outI(j, 1) = a(3)
            outJ(j, 1) = a(4)
            outK(j, 1) = a(5)
            outL(j, 1) = a(6)
            outM(j, 1) = dia
            outN(j, 1) = fecha
            outO(j, 1) = a(9)
            outR(j, 1) = a(10)
            outS(j, 1) = a(11)
            outT(j, 1) = a(12)

            outU(j, 1) = nextID
            nextID = nextID + 1

            outW(j, 1) = modUID.BuildUID_Incidencia(gLoc, empNum, gAnio, gMes, gTipoPeriodo, gPeriodo, dia)
        Next j

        wsBD.Range("A" & startRow).Resize(n, 1).value = outA
        wsBD.Range("C" & startRow).Resize(n, 1).value = outC
        wsBD.Range("I" & startRow).Resize(n, 1).value = outI
        wsBD.Range("J" & startRow).Resize(n, 1).value = outJ
        wsBD.Range("K" & startRow).Resize(n, 1).value = outK
        wsBD.Range("L" & startRow).Resize(n, 1).value = outL
        wsBD.Range("M" & startRow).Resize(n, 1).value = outM
        wsBD.Range("N" & startRow).Resize(n, 1).value = outN
        wsBD.Range("O" & startRow).Resize(n, 1).value = outO
        wsBD.Range("R" & startRow).Resize(n, 1).value = outR
        wsBD.Range("S" & startRow).Resize(n, 1).value = outS
        wsBD.Range("T" & startRow).Resize(n, 1).value = outT
        wsBD.Range("U" & startRow).Resize(n, 1).value = outU
        wsBD.Range("W" & startRow).Resize(n, 1).value = outW
    End If

    '========================================
    ' 8) Regenerar matriz (para ver cambios ya)
    '========================================
    On Error Resume Next
    modReporteIncidencias.GenerarMatrizPeriodoActual
    On Error GoTo 0

    '-----------------------------------------
    ' 9) Reproteger BD (opcional)
    '-----------------------------------------
    On Error Resume Next
    wsBD.Protect Password:="AVASA"
    On Error GoTo 0

    '-----------------------------------------
    ' 10) Restore
    '-----------------------------------------
    Application.StatusBar = False
    Application.Calculation = calc0
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True

    MsgBox "Precarga OK" & vbCrLf & _
           "Insertados: " & countInsert & vbCrLf & _
           "Actualizados (checador): " & countUpdateChecador & vbCrLf & _
           "Rellenados (vacío->checador): " & countFillEmpty & vbCrLf & _
           "Saltados (manual): " & countSkipManual & vbCrLf & _
           "Saltados (UserID no numérico): " & countSkipBadEmp, vbInformation

    Exit Sub
LogEnd "PrecargarChecador"
Fail:
    On Error Resume Next
    Application.StatusBar = False
    Application.Calculation = xlCalculationAutomatic
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    MsgBox "Error en precarga: " & err.Number & " - " & err.Description, vbCritical
End Sub

Private Function CodigoAsistenciaPorFecha(ByVal f As Date) As String
    If Weekday(f, vbMonday) = 7 Then
        CodigoAsistenciaPorFecha = "PD"
    ElseIf EsFestivo_Local(f) Then
        CodigoAsistenciaPorFecha = "DF"
    Else
        CodigoAsistenciaPorFecha = "X"
    End If
End Function

Private Function EsFestivo_Local(ByVal f As Date) As Boolean
    On Error Resume Next
    EsFestivo_Local = modCalendario.EsFestivo(f)
End Function

Private Function NormalizarTipoPeriodo(ByVal v As Variant) As String
    Dim s As String
    s = UCase$(Trim$(CStr(v)))

    If s = "QUINCENAL" Or s = "Q" Or Left$(s, 1) = "Q" Then
        NormalizarTipoPeriodo = "QUINCENAL"
    ElseIf s = "SEMANAL" Or s = "S" Or Left$(s, 1) = "S" Then
        NormalizarTipoPeriodo = "SEMANAL"
    Else
        NormalizarTipoPeriodo = s
    End If
End Function

Private Function GetNextID_BD(ByVal wsBD As Worksheet) As Long
    On Error GoTo EH
    Dim lastRow As Long, mx As Long, v As Variant, i As Long

    lastRow = wsBD.Cells(wsBD.Rows.count, "U").End(xlUp).Row
    mx = 0
    If lastRow < 2 Then
        GetNextID_BD = 1
        Exit Function
    End If

    For i = 2 To lastRow
        v = wsBD.Cells(i, "U").value
        If IsNumeric(v) Then
            If CLng(v) > mx Then mx = CLng(v)
        End If
    Next i

    GetNextID_BD = mx + 1
    Exit Function

EH:
    GetNextID_BD = 1
End Function

'--- Helper: obtener ListObject por hoja+tabla (sin choque de nombres)
Private Function GetLO(ByVal wsName As String, ByVal loName As String) As ListObject
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(wsName)
    On Error GoTo 0
    If ws Is Nothing Then err.Raise vbObjectError + 501, "GetLO", "No existe la hoja: " & wsName

    On Error Resume Next
    Set GetLO = ws.ListObjects(loName)
    On Error GoTo 0
    If GetLO Is Nothing Then err.Raise vbObjectError + 502, "GetLO", "No existe la tabla: " & loName & " en hoja " & wsName
End Function

Private Function NzStr(ByVal v As Variant) As String
    If IsError(v) Or IsNull(v) Or IsEmpty(v) Then
        NzStr = ""
    Else
        NzStr = Trim$(CStr(v))
    End If
End Function

'--- IMPORTANTE: esto lo usa modReporteIncidencias (y temporales)
Public Function UClean(ByVal v As Variant) As String
    If IsError(v) Or IsNull(v) Or IsEmpty(v) Then
        UClean = ""
    Else
        UClean = UCase$(Trim$(CStr(v)))
    End If
End Function

Public Function GetCatalogoPuestosCanon() As Variant
    Dim loTbl As ListObject, arr(), i As Long, out As Collection
    Dim idxP As Long, idxAct As Long
    Dim p As String, activo As Long

    Set out = New Collection
    Set loTbl = GetLO("CAT_PUESTOS", "tblPuestosCanon")

    idxP = loTbl.ListColumns("PuestoCanon").Index
    idxAct = loTbl.ListColumns("Activo").Index

    For i = 1 To loTbl.ListRows.count
        p = UClean(loTbl.DataBodyRange.Cells(i, idxP).value)
        activo = CLng(val(loTbl.DataBodyRange.Cells(i, idxAct).value))
        If Len(p) > 0 And activo <> 0 Then out.Add p
    Next i

    If out.count = 0 Then
        GetCatalogoPuestosCanon = Array("")
        Exit Function
    End If

    ReDim arr(0 To out.count - 1)
    For i = 1 To out.count
        arr(i - 1) = out(i)
    Next i

    GetCatalogoPuestosCanon = arr
End Function

'====================================================
' Empleados eliminados por periodo (para NO completar)
' Hoja: Empleados_Eliminados (VeryHidden)
'
' Columnas:
' A: Loc
' B: PeriodoKey   (LOC|YYYY|MM|TIPO|N)
' C: NumEmpleado
' D: Nombre (opcional)
' E: Motivo (opcional)
' F: Usuario
' G: FechaHora

Private Const SH_EMPLEADOS_ELIMINADOS As String = "Empleados_Eliminados"

Private Function PeriodoKeyActual() As String
    PeriodoKeyActual = UCase$(Trim$(gLoc)) & "|" & gAnio & "|" & Format$(gMes, "00") & "|" & _
                       UCase$(NormalizarTipoPeriodo(gTipoPeriodo)) & "|" & gPeriodo
End Function

Public Function GetWsEmpleadosEliminados(Optional ByVal crearSiNoExiste As Boolean = True) As Worksheet
    On Error Resume Next
    Set GetWsEmpleadosEliminados = ThisWorkbook.Worksheets(SH_EMPLEADOS_ELIMINADOS)
    On Error GoTo 0

    If GetWsEmpleadosEliminados Is Nothing And crearSiNoExiste Then
        CrearHojaEmpleadosEliminados
        Set GetWsEmpleadosEliminados = ThisWorkbook.Worksheets(SH_EMPLEADOS_ELIMINADOS)
    End If
End Function

Private Sub CrearHojaEmpleadosEliminados()
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets(SH_EMPLEADOS_ELIMINADOS)
    On Error GoTo 0
    If Not ws Is Nothing Then Exit Sub

    Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.count))
    ws.Name = SH_EMPLEADOS_ELIMINADOS

    ws.Range("A1").value = "Loc"
    ws.Range("B1").value = "PeriodoKey"
    ws.Range("C1").value = "NumEmpleado"
    ws.Range("D1").value = "Nombre"
    ws.Range("E1").value = "Motivo"
    ws.Range("F1").value = "Usuario"
    ws.Range("G1").value = "FechaHora"

    ws.Rows(1).Font.Bold = True
    ws.Columns("A:G").EntireColumn.AutoFit

    ws.Visible = xlSheetVeryHidden
End Sub

Public Function EsEmpleadoEliminadoPeriodo(ByVal numEmp As Long, Optional ByVal periodoKey As String = "") As Boolean
    On Error GoTo salir

    Dim ws As Worksheet, lastRow As Long, r As Long
    Dim key As String

    If periodoKey = "" Then periodoKey = PeriodoKeyActual()
    key = UCase$(Trim$(periodoKey))

    Set ws = GetWsEmpleadosEliminados(False)
    If ws Is Nothing Then GoTo salir

    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then GoTo salir

    For r = 2 To lastRow
        If UCase$(Trim$(CStr(ws.Cells(r, "A").value))) = UCase$(Trim$(gLoc)) _
           And UCase$(Trim$(CStr(ws.Cells(r, "B").value))) = key _
           And CLng(val(ws.Cells(r, "C").value)) = CLng(numEmp) Then
            EsEmpleadoEliminadoPeriodo = True
            Exit Function
        End If
    Next r

salir:
    EsEmpleadoEliminadoPeriodo = False
End Function

Public Sub MarcarEmpleadoEliminadoPeriodo( _
    ByVal numEmp As Long, _
    Optional ByVal nombre As String = "", _
    Optional ByVal motivo As String = "" _
)
    Dim ws As Worksheet
    Dim lastRow As Long, r As Long
    Dim key As String

    key = PeriodoKeyActual()
    Set ws = GetWsEmpleadosEliminados(True)

    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then lastRow = 1

    'si ya existe -> update
    For r = 2 To lastRow
        If UCase$(Trim$(CStr(ws.Cells(r, "A").value))) = UCase$(Trim$(gLoc)) _
           And UCase$(Trim$(CStr(ws.Cells(r, "B").value))) = UCase$(Trim$(key)) _
           And CLng(val(ws.Cells(r, "C").value)) = CLng(numEmp) Then

            If Len(Trim$(nombre)) > 0 Then ws.Cells(r, "D").value = nombre
            If Len(Trim$(motivo)) > 0 Then ws.Cells(r, "E").value = motivo
            ws.Cells(r, "F").value = Environ$("username")
            ws.Cells(r, "G").value = Now
            Exit Sub
        End If
    Next r

    'insert
    r = lastRow + 1
    ws.Cells(r, "A").value = gLoc
    ws.Cells(r, "B").value = key
    ws.Cells(r, "C").value = CLng(numEmp)
    ws.Cells(r, "D").value = nombre
    ws.Cells(r, "E").value = motivo
    ws.Cells(r, "F").value = Environ$("username")
    ws.Cells(r, "G").value = Now
End Sub

'Quita la marca de eliminado (por si lo regresan)
Public Sub DesmarcarEmpleadoEliminadoPeriodo(ByVal numEmp As Long)
    Dim ws As Worksheet
    Dim lastRow As Long, r As Long
    Dim key As String

    key = PeriodoKeyActual()
    Set ws = GetWsEmpleadosEliminados(False)
    If ws Is Nothing Then Exit Sub

    lastRow = ws.Cells(ws.Rows.count, "A").End(xlUp).Row
    If lastRow < 2 Then Exit Sub

    For r = lastRow To 2 Step -1
        If UCase$(Trim$(CStr(ws.Cells(r, "A").value))) = UCase$(Trim$(gLoc)) _
           And UCase$(Trim$(CStr(ws.Cells(r, "B").value))) = UCase$(Trim$(key)) _
           And CLng(val(ws.Cells(r, "C").value)) = CLng(numEmp) Then
            ws.Rows(r).Delete
            Exit Sub
        End If
    Next r
End Sub

Public Sub CacheUnicosPuestoActividad_Global(ByVal loDB As Object)

    Dim ws As Worksheet
    Dim dictP As Object, dictA As Object
    Dim i As Long
    Dim idxPuesto As Long, idxAct As Long
    Dim v As String
    Dim r As Long

    Set dictP = CreateObject("Scripting.Dictionary")
    Set dictA = CreateObject("Scripting.Dictionary")
    dictP.CompareMode = vbTextCompare
    dictA.CompareMode = vbTextCompare

    On Error GoTo EH

    idxPuesto = loDB.ListColumns("PUESTO").Index
    idxAct = loDB.ListColumns("ACTIVIDAD").Index

    'Crear/obtener hoja cache
    On Error Resume Next
    Set ws = ThisWorkbook.Worksheets("ETL_Uni_PuestoActividad")
    On Error GoTo 0

        Set ws = ThisWorkbook.Worksheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.count))

        r = r + 1
    Next k

Public Sub AutoFix_ConfigPaths()
    On Error GoTo EH

    Dim rootGerentes As String
    rootGerentes = GetGerentesRootPath_FromWorkbookPath(ThisWorkbook.path)
    If Len(rootGerentes) = 0 Then Exit Sub

    Dim masterDataFolder As String
    masterDataFolder = rootGerentes & "\_MASTERDATA"

    Dim empDbDefault As String
    empDbDefault = masterDataFolder & "\Base de datos empleados.xlsx"

    '1) Guardar root (y compatibilidad)
    Dim curRoot As String
    curRoot = Trim$(GetConfig("GerentesRootPath", ""))
    If NeedsFix(curRoot) Then SetConfig "GerentesRootPath", rootGerentes

    'MasterDBPath históricamente lo estás usando como ROOT
    Dim curMaster As String
    curMaster = Trim$(GetConfig("MasterDBPath", ""))
    If NeedsFix(curMaster) Then SetConfig "MasterDBPath", rootGerentes

    '2) EmployeeDBPath: si trae OPERADOR1 o no existe, se repara
    Dim curEmp As String
    curEmp = Trim$(GetConfig("EmployeeDBPath", ""))

    If NeedsFix(curEmp) Then
        'Primero: si existe una copia local junto al archivo, úsala
        Dim localEmp As String
        localEmp = ThisWorkbook.path & "\Base de datos empleados.xlsx"
        If FileExists(localEmp) Then
            SetConfig "EmployeeDBPath", localEmp
        Else
            SetConfig "EmployeeDBPath", empDbDefault
        End If
    End If

    Exit Sub

EH:
    'No bloqueamos apertura por fallos de autofix
End Sub

Private Function NeedsFix(ByVal p As String) As Boolean
    If Len(p) = 0 Then NeedsFix = True: Exit Function
    If InStr(1, p, "\OPERADOR1\", vbTextCompare) > 0 Then NeedsFix = True: Exit Function
    If InStr(1, p, "/personal/", vbTextCompare) > 0 Then NeedsFix = True: Exit Function
    If LCase$(Left$(p, 4)) = "http" Then NeedsFix = True: Exit Function
    If Not PathExists(p) Then NeedsFix = True: Exit Function
    NeedsFix = False
End Function

Private Function PathExists(ByVal p As String) As Boolean
    On Error Resume Next
    If Len(p) = 0 Then PathExists = False: Exit Function

    If LCase$(Right$(p, 5)) = ".xlsx" Or LCase$(Right$(p, 5)) = ".xlsm" Then
        PathExists = (Len(Dir(p)) > 0)
    Else
        PathExists = (Len(Dir(p, vbDirectory)) > 0)
    End If
    On Error GoTo 0
End Function

Private Function FileExists(ByVal p As String) As Boolean
    On Error Resume Next
    FileExists = (Len(Dir(p)) > 0)
    On Error GoTo 0
End Function

Public Function GetGerentesRootPath_FromWorkbookPath(ByVal wbPath As String) As String
    'Esperado:
    '...\REPORTES GERENTES\<LOC>\REPORTE DE INCIDENCIAS DE NOMINA
    'Subimos 2 niveles: (REPORTE...) -> (LOC) -> (REPORTES GERENTES)

    Dim p As String
    p = wbPath

    p = ParentFolder(p) '...\<LOC>
    If Len(p) = 0 Then Exit Function

    p = ParentFolder(p) '...\REPORTES GERENTES
    If Len(p) = 0 Then Exit Function

    GetGerentesRootPath_FromWorkbookPath = p
End Function

Private Function ParentFolder(ByVal path As String) As String
    Dim i As Long
    path = Replace(path, "/", "\")
    If Right$(path, 1) = "\" Then path = Left$(path, Len(path) - 1)
    i = InStrRev(path, "\")
    If i <= 0 Then ParentFolder = "": Exit Function
    ParentFolder = Left$(path, i - 1)
End Function

Private Const LOCAL_SHEET As String = "Empleados"


Public Sub Refresh_Empleados_Todos()
    On Error GoTo EH

    Dim locCode As String
    locCode = Trim$(GetConfig("LocationCode", ""))

    If Len(locCode) = 0 Then
        MsgBox "No encuentro LocationCode en Config.", vbCritical
        Exit Sub
    End If

    Dim logPath As String
    logPath = Environ$("TEMP") & "\REFRESH_EMPLEADOS_" & locCode & "_" & Format(Now, "yyyymmdd_hhnnss") & ".txt"

    Dim nOk As Long, nSkip As Long, nErr As Long
    nOk = 0: nSkip = 0: nErr = 0

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False
    Application.StatusBar = "Refrescando empleados (" & locCode & ")..."

    LogLine logPath, "=== REFRESH EMPLEADOS ==="
    LogLine logPath, "LocCode: " & locCode

    Dim empPath As String
    empPath = ResolveEmployeeDBPath()

    LogLine logPath, "EmployeeDBPath(resuelto): " & empPath

    If Len(empPath) = 0 Or Len(Dir(empPath)) = 0 Then
        nSkip = nSkip + 1
        LogLine logPath, "SKIP | No existe BD empleados en ruta resuelta."
        MsgBox "No pude encontrar la base de empleados para actualizar." & vbCrLf & vbCrLf & _
               "El archivo puede seguir operando con la tabla local, pero la lista podría estar desactualizada." & vbCrLf & _
               "Ruta intentada: " & empPath, vbExclamation
        GoTo SAFEEXIT
    End If

    Dim wbSrc As Workbook, wsSrc As Worksheet, loSrc As ListObject
    Dim wbWasOpened As Boolean
    wbWasOpened = False

    'Abrir source read-only
    Set wbSrc = Workbooks.Open(fileName:=empPath, ReadOnly:=True, UpdateLinks:=False)
    wbWasOpened = True

    Dim tblName As String
    tblName = Trim$(GetConfig("EmployeeDBTable", MASTER_TABLE_NAME_DEFAULT))
    If Len(tblName) = 0 Then tblName = MASTER_TABLE_NAME_DEFAULT

    Set loSrc = FindTableByName(wbSrc, tblName)
    If loSrc Is Nothing Then
        nErr = nErr + 1
        LogLine logPath, "ERR | No encontré tabla '" & tblName & "' en BD."
        MsgBox "No encontré la tabla '" & tblName & "' dentro de la base de empleados.", vbCritical
        GoTo SAFEEXIT
    End If

    Dim arrData As Variant, arrHeaders As Variant
    arrHeaders = loSrc.HeaderRowRange.value
    arrData = GetFilteredTableData(loSrc, MASTER_COL_LOCCODE, locCode)

    If IsEmpty(arrData) Then
        LogLine logPath, "OK | 0 filas para locación " & locCode
        'Dejar tabla local vacía (con headers)
        ReplaceLocalTableData arrHeaders, Empty
        nOk = nOk + 1
        GoTo SAFEEXIT
    End If

    ReplaceLocalTableData arrHeaders, arrData
    nOk = nOk + 1

    LogLine logPath, "OK | Refresh completado. Filas: " & UBound(arrData, 1)

SAFEEXIT:
    On Error Resume Next
    If wbWasOpened Then wbSrc.Close SaveChanges:=False

    Application.StatusBar = False
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    On Error GoTo 0

    MsgBox "Refresh terminado." & vbCrLf & _
           "OK=" & nOk & " | SKIP=" & nSkip & " | ERR=" & nErr & vbCrLf & vbCrLf & _
           "LOG: " & logPath, vbInformation

    Exit Sub

EH:
    Application.StatusBar = False
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    MsgBox "ERROR Refresh_Empleados_Todos: " & err.Number & " - " & err.Description, vbCritical
End Sub

'-------------------------
' Resuelve ruta de BD empleados sin OPERADOR1
'-------------------------
Private Function ResolveEmployeeDBPath() As String
    Dim p As String
    p = Trim$(GetConfig("EmployeeDBPath", ""))

    'Si la config ya sirve, úsala
    If Len(p) > 0 Then
        If LCase$(Left$(p, 4)) <> "http" And Len(Dir(p)) > 0 Then
            ResolveEmployeeDBPath = p
            Exit Function
        End If
    End If

    '1) Intentar copia local junto al archivo
    Dim localEmp As String
    localEmp = ThisWorkbook.path & "\Base de datos empleados.xlsx"
    If Len(Dir(localEmp)) > 0 Then
        ResolveEmployeeDBPath = localEmp
        Exit Function
    End If

    '2) Intentar root\_MASTERDATA\Base de datos empleados.xlsx
    Dim rootGerentes As String
    rootGerentes = Trim$(GetConfig("GerentesRootPath", ""))
    If Len(rootGerentes) = 0 Then rootGerentes = GetGerentesRootPath_FromWorkbookPath(ThisWorkbook.path)

    If Len(rootGerentes) > 0 Then
        Dim p2 As String
        p2 = rootGerentes & "\_MASTERDATA\Base de datos empleados.xlsx"
        If Len(Dir(p2)) > 0 Then
            ResolveEmployeeDBPath = p2
            Exit Function
        End If
    End If

    ResolveEmployeeDBPath = p
End Function

    For Each ws In wb.Worksheets
        For Each lo In ws.ListObjects
            If StrComp(lo.Name, tableName, vbTextCompare) = 0 Then

                Exit Function
            End If
        Next lo

'-------------------------
' Filtra ListObject por columna = valor y regresa array (sin headers)
'-------------------------
Private Function GetFilteredTableData(ByVal lo As ListObject, ByVal colName As String, ByVal matchValue As String) As Variant
    On Error GoTo EH

    Dim idx As Long
    idx = lo.ListColumns(colName).Index

    Dim src As Variant
    src = lo.DataBodyRange.value

    Dim i As Long, j As Long, outRow As Long
    Dim outArr() As Variant
    Dim nCols As Long
    nCols = UBound(src, 2)

    'Primero contar
    Dim cnt As Long
    cnt = 0
    For i = 1 To UBound(src, 1)
        If Trim$(CStr(src(i, idx))) = matchValue Then
            cnt = cnt + 1
        End If
    Next i

    If cnt = 0 Then
        GetFilteredTableData = Empty
        Exit Function
    End If

    ReDim outArr(1 To cnt, 1 To nCols)
    outRow = 0

    For i = 1 To UBound(src, 1)
        If Trim$(CStr(src(i, idx))) = matchValue Then
            outRow = outRow + 1
            For j = 1 To nCols
                outArr(outRow, j) = src(i, j)
            Next j
        End If
    Next i

    GetFilteredTableData = outArr
    Exit Function

EH:
    GetFilteredTableData = Empty
End Function

Private Function ResolveLocalEmpTableName(ByVal ws As Worksheet) As String
    'Prioridad: tblEmpleados (la que usa Push) y si no existe, tblEmpleados_Local (legacy)
    On Error Resume Next

    Dim lo As ListObject

    Set lo = ws.ListObjects("tblEmpleados")
    If Not lo Is Nothing Then
        ResolveLocalEmpTableName = "tblEmpleados"
        Exit Function
    End If

    Set lo = Nothing
    Set lo = ws.ListObjects("tblEmpleados_Local")
    If Not lo Is Nothing Then
        ResolveLocalEmpTableName = "tblEmpleados_Local"
        Exit Function
    End If

    ResolveLocalEmpTableName = vbNullString
End Function


'-------------------------
' Reemplaza SOLO la data de tblEmpleados_Local (sin tocar nada más)
'-------------------------
Private Sub ReplaceLocalTableData(ByVal headers As Variant, ByVal data As Variant)
    On Error GoTo EH

Dim ws As Worksheet, lo As ListObject
Set ws = ThisWorkbook.Worksheets(LOCAL_SHEET)

Dim localTableName As String
localTableName = ResolveLocalEmpTableName(ws)

If Len(localTableName) = 0 Then
    err.Raise vbObjectError + 510, , "No existe tabla local de empleados en hoja '" & LOCAL_SHEET & _
                                     "'. Esperaba 'tblEmpleados' o 'tblEmpleados_Local'."
End If

Set lo = ws.ListObjects(localTableName)


    'Validar headers: que coincidan columnas
    Dim nCols As Long
    nCols = lo.ListColumns.count

    'Borrar data actual
    If Not lo.DataBodyRange Is Nothing Then
        lo.DataBodyRange.Delete
    End If

    'Si no hay data, deja tabla vacía (solo headers)
    If IsEmpty(data) Then Exit Sub

    Dim nRows As Long
    nRows = UBound(data, 1)

    'Insertar filas necesarias
    Dim i As Long
    For i = 1 To nRows
        lo.ListRows.Add
    Next i

    'Pegar valores
    lo.DataBodyRange.value = data
    Exit Sub

EH:
    MsgBox "ERROR ReplaceLocalTableData: " & err.Number & " - " & err.Description, vbCritical
End Sub

Option Explicit

'========================

Public Const MASTER_TABLE_DEFAULT As String = "tblEmpleados"
Public Const MASTER_LOC_COL_DEFAULT As String = "LocCode"

Private Const SHEET_LOC As String = "Locaciones"
Private Const TBL_LOC As String = "tblLocaciones"
Private Const COL_ACTIVE As String = "Active"
Private Const COL_LOCCODE As String = "LocationCode"

Private Const LOCAL_EMP_SHEET As String = "Empleados"
Private Const LOCAL_EMP_TABLE As String = "tblEmpleados"




'========================
' ENTRYPOINT
'========================
Public Sub PushEmpleados_A_TodasLasLocaciones()
    On Error GoTo EH

    Dim basePath As String, masterPath As String
    basePath = CleanPath(GetConfigValue("GerentesRootPath", ""))
    masterPath = CleanPath(GetConfigValue("MasterDBPath", ""))


    If Len(basePath) = 0 Then err.Raise vbObjectError + 1, , "Config: falta GerentesRootPath"
    If Right$(basePath, 1) <> "\" Then basePath = basePath & "\"

    If Len(masterPath) = 0 Then err.Raise vbObjectError + 2, , "Config: falta MasterDBPath"
    If Len(Dir(masterPath)) = 0 Then err.Raise vbObjectError + 3, , "No existe MasterDBPath: " & masterPath

    Dim masterTableName As String, masterLocCol As String
    masterTableName = Trim$(GetConfigValue("EmployeeDBTable", MASTER_TABLE_DEFAULT))
    If Len(masterTableName) = 0 Then masterTableName = MASTER_TABLE_DEFAULT

    masterLocCol = Trim$(GetConfigValue("EmployeeDBLocColumn", MASTER_LOC_COL_DEFAULT))
    If Len(masterLocCol) = 0 Then masterLocCol = MASTER_LOC_COL_DEFAULT

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False

    '1) Abrir Master una sola vez
    Dim wbMaster As Workbook
    Set wbMaster = Workbooks.Open(fileName:=masterPath, ReadOnly:=True, UpdateLinks:=False)

    Dim loMaster As ListObject
    Set loMaster = FindTableByName(wbMaster, masterTableName)
    If loMaster Is Nothing Then err.Raise vbObjectError + 10, , "No encontré tabla '" & masterTableName & "' en MasterDB."

    Dim idxLoc As Long
    idxLoc = loMaster.ListColumns(masterLocCol).Index

    '2) Locaciones
    Dim loLoc As ListObject
    Set loLoc = ThisWorkbook.Worksheets(SHEET_LOC).ListObjects(TBL_LOC)

    Dim idxActive As Long, idxCode As Long
    idxActive = loLoc.ListColumns(COL_ACTIVE).Index
    idxCode = loLoc.ListColumns(COL_LOCCODE).Index

    Dim r As ListRow
    Dim locCode As String, locFile As String
    Dim okCount As Long, errCount As Long
    okCount = 0: errCount = 0

    For Each r In loLoc.ListRows
        If val(r.Range.Cells(1, idxActive).value) = 1 Then
            locCode = Trim$(CStr(r.Range.Cells(1, idxCode).value))
            If Len(locCode) = 0 Then GoTo NEXT_LOC

            locFile = basePath & locCode & "\REPORTE DE INCIDENCIAS DE NOMINA\Incidencias_" & locCode & ".xlsm"
            If Len(Dir(locFile)) = 0 Then
                errCount = errCount + 1
                GoTo NEXT_LOC
            End If

            On Error GoTo EH_LOC

            '3) Filtrar empleados de esa locación (desde master abierto)
            Dim arrData As Variant
            arrData = GetFilteredTableData_Fast(loMaster, idxLoc, locCode)

            '4) Abrir archivo de locación e inyectar
            Dim wbLoc As Workbook
            Set wbLoc = Workbooks.Open(fileName:=locFile, UpdateLinks:=False)

            ReplaceLocalTableData_InWorkbook wbLoc, arrData

            wbLoc.Close SaveChanges:=True
            Set wbLoc = Nothing

            okCount = okCount + 1
        End If

NEXT_LOC:
        On Error GoTo EH
        DoEvents
    Next r

SAFE_EXIT:
    On Error Resume Next
    wbMaster.Close SaveChanges:=False
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    MsgBox "Push terminado." & vbCrLf & "OK=" & okCount & " | ERR=" & errCount, vbInformation
    Exit Sub

EH_LOC:
    errCount = errCount + 1
    On Error Resume Next
    If Not wbLoc Is Nothing Then wbLoc.Close SaveChanges:=False
    Resume NEXT_LOC

EH:
    MsgBox "Error: " & err.Number & vbCrLf & err.Description, vbCritical
    Resume SAFE_EXIT
End Sub

'Filtra por valor en una columna (idxCol) SIN AutoFilter (rápido y estable)

    cnt = 0
    For i = 1 To UBound(src, 1)
        If Trim$(CStr(src(i, idxCol))) = matchValue Then cnt = cnt + 1
    Next i

    If cnt = 0 Then
        GetFilteredTableData_Fast = Empty
        Exit Function
    End If

    Dim outArr() As Variant
    ReDim outArr(1 To cnt, 1 To nCols)

    outRow = 0
    For i = 1 To UBound(src, 1)
        If Trim$(CStr(src(i, idxCol))) = matchValue Then
            outRow = outRow + 1
            For j = 1 To nCols
                outArr(outRow, j) = src(i, j)
            Next j
        End If
    Next i

    GetFilteredTableData_Fast = outArr
    Exit Function

EH:
    GetFilteredTableData_Fast = Empty
End Function

Private Function CleanPath(ByVal p As String) As String
    p = Trim$(p)
    p = Replace(p, Chr$(34), "") ' quita comillas "
    p = Replace(p, "/", "\")     ' normaliza slash
    Do While InStr(p, "\\") > 0

'? helper PRIVADO (no choca con otros CleanPath)
Private Function GT_CleanPath(ByVal p As String) As String
    p = Trim$(p)
    p = Replace(p, Chr$(34), "")
    p = Replace(p, "/", "\")
    Do While InStr(p, "\\") > 0
        p = Replace(p, "\\", "\")
    Loop
    GT_CleanPath = p
End Function

Option Explicit

'=========================
' LOG SILENCIOSO AVASA
' Ruta fija: C:\AVASA_TMP\_LOGS
' Fallback: %TEMP%
' Invisible para gerentes
'=========================

Private Const LOG_ENABLED As Boolean = True
Private Const LOG_BASE_PATH As String = "C:\AVASA_TMP\_LOGS"

Public Sub LogStart(ByVal scope As String)
    LogWrite "START", scope, ""
End Sub

Public Sub LogEnd(ByVal scope As String)
    LogWrite "END", scope, ""
End Sub

Public Sub LogInfo(ByVal msg As String, Optional ByVal scope As String = "")
    LogWrite "INFO", scope, msg
End Sub

Public Sub LogWarn(ByVal msg As String, Optional ByVal scope As String = "")
    LogWrite "WARN", scope, msg
End Sub

Public Sub LogErr(ByVal msg As String, ByVal er As ErrObject, Optional ByVal scope As String = "")
    Dim fullMsg As String
    fullMsg = msg & " | Err=" & CStr(er.Number) & " | " & er.Description
    LogWrite "ERR", scope, fullMsg
End Sub

Public Sub LogKV(ByVal k As String, ByVal v As Variant, Optional ByVal scope As String = "")
    LogWrite "KV", scope, k & "=" & SafeStr(v)
End Sub

'-------------------------
' Internals
'-------------------------
Private Sub LogWrite(ByVal level As String, ByVal scope As String, ByVal msg As String)
    If Not LOG_ENABLED Then Exit Sub
    
    Dim line As String
    line = Format$(Now, "yyyy-mm-dd hh:nn:ss") _
        & " | " & level _
        & " | " & IIf(Len(scope) > 0, scope, "-") _
        & " | " & msg
    
    AppendLineToFile GetLogFilePath(), line
End Sub

Private Function GetLogFilePath() As String
    On Error GoTo fallback
    
    EnsureFolder LOG_BASE_PATH
    
    ' Un log por día
    GetLogFilePath = LOG_BASE_PATH & "\LOG_" & Format$(Date, "yyyymmdd") & ".txt"
    Exit Function
    
fallback:
    ' Fallback extremo
    Dim tmp As String
    tmp = Environ$("TEMP")
    If Len(tmp) = 0 Then tmp = CurDir$
    GetLogFilePath = tmp & "\AVASA_LOG_" & Format$(Date, "yyyymmdd") & ".txt"
End Function

Private Sub EnsureFolder(ByVal folderPath As String)
    On Error Resume Next
    If Dir(folderPath, vbDirectory) = vbNullString Then
        MkDir folderPath
    End If
    On Error GoTo 0
End Sub

Private Sub AppendLineToFile(ByVal filePath As String, ByVal line As String)
    On Error Resume Next
    
    Dim ff As Integer
    ff = FreeFile
    Open filePath For Append As #ff
    Print #ff, line
    Close #ff
    
    On Error GoTo 0
End Sub

Private Function SafeStr(ByVal v As Variant) As String
    On Error GoTo safe
    If IsError(v) Then
        SafeStr = "#ERROR"
    ElseIf IsNull(v) Then
        SafeStr = "NULL"
    Else
        SafeStr = CStr(v)
    End If
    Exit Function
safe:
    SafeStr = "<?>"
End Function
